# Nation Atlas: whether a level ladder button can be used (country scope)

mtui_lp_BUOffice_up_ok = {
	custom_trigger_tooltip = {
		tooltip = Pow_BUChange
		NOT = { has_country_modifier = Prov_BUBlock }
	}
	if = {
		limit = { Privilege_Has = { Privilege = BUOffice Lvl = 0 } }
		Privilege_BUOffice1CanRaise = yes
	}
	else_if = {
		limit = { Privilege_Has = { Privilege = BUOffice Lvl = 1 } }
		Privilege_BUOffice2CanRaise = yes
	}
	else = { custom_trigger_tooltip = { tooltip = mtui_st_top always = no } }
}
