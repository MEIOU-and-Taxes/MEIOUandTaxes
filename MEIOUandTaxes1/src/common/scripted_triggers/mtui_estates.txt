# An estate exists in the realm while it has power or loyalty (clans in a realm without them have neither)
mtui_estate_exists_NO = {
	custom_trigger_tooltip = {
		tooltip = mtui_estate_exists_tt
		OR = {
			NOT = { is_key_equal = { lhs = Prov_NOPow value = 0 } }
			NOT = { is_key_equal = { lhs = Prov_NOLoy value = 0 } }
		}
	}
}
