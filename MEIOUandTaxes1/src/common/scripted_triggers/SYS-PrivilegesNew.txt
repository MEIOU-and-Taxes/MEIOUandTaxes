Privilege_Has = {
	custom_trigger_tooltip = {
		tooltip = Privilege_$Privilege$$Lvl$Has_tt
		is_key_equal = { lhs = Privilege_$Privilege$ value = $Lvl$ }
	}
}
Privilege_HasGreater = {
	custom_trigger_tooltip = {
		tooltip = Privilege_$Privilege$$Lvl$HasGreater_tt
		check_key = { lhs = Privilege_$Privilege$ value = $Lvl$ }
	}
}
Privilege_NotHas = {
	custom_trigger_tooltip = {
		tooltip = Privilege_$Privilege$$Lvl$NotHas_tt
		NOT = { is_key_equal = { lhs = Privilege_$Privilege$ value = $Lvl$ } }
	}
}
Privilege_NotHasGreater = {
	custom_trigger_tooltip = {
		tooltip = Privilege_$Privilege$$Lvl$NotHasGreater_tt
		NOT = { check_key = { lhs = Privilege_$Privilege$ value = $Lvl$ } }
	}
}
