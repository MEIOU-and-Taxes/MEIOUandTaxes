# Province view: build panel shortcuts
custom_button = {
	name = mtui_bw_build_capitol
	potential = { OR = { owned_by = FROM owner = { is_subject_of = FROM } } }
	trigger = {
		custom_trigger_tooltip = {
			tooltip = mtui_bw_needs_city
			check_key = { lhs = RE_Total value = 0.001 }
		}
	}
	effect = {
		hidden_effect = { Pow_UI = yes }
		owner = {
			hidden_effect = {
				set_key = { lhs = Construct_Building value = 5 }
				set_key = { lhs = Construct_TypeToggle value = 1 }
				set_key = { lhs = Construct_Amount value = 1 }
				set_key = { lhs = Construct_Type value = 3 }
			}
			country_event = { id = SYS_Construct.001 }
		}
	}
	tooltip = mtui_bw_build_capitol_tt
}

custom_button = {
	name = mtui_bw_invest_agriculture
	potential = { OR = { owned_by = FROM owner = { is_subject_of = FROM } } }
	trigger = {
		custom_trigger_tooltip = {
			tooltip = mtui_bw_req_money
			FROM = {
				OR = {
					AND = { has_country_flag = mtui_invest_10 treasury = 10 }
					AND = { has_country_flag = mtui_invest_50 treasury = 50 }
					AND = { has_country_flag = mtui_invest_100 treasury = 100 }
					AND = { has_country_flag = mtui_invest_250 treasury = 250 }
					AND = { has_country_flag = mtui_invest_500 treasury = 500 }
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = mtui_bw_req_reach
			check_key = { lhs = Prov_BUPow value = 2.5 }
		}
		custom_trigger_tooltip = {
			tooltip = mtui_bw_req_exists
			check_key = { lhs = Building_Agriculture value = 0.001 }
		}
		custom_trigger_tooltip = {
			tooltip = mtui_bw_req_max
			NOT = { is_key_equal = { lhs = Building_Agriculture which = Building_AgricultureMax } }
		}
		custom_trigger_tooltip = {
			tooltip = mtui_bw_req_control
			OR = { owned_by = FROM owner = { is_subject_of = FROM } }
			controlled_by = owner
		}
	}
	effect = {
		if = {
			limit = { FROM = { has_country_flag = mtui_invest_10 } }
			change_key = { lhs = Building_AgricultureWealth value = 10 }
			FROM = { add_treasury = -10 }
		}
		if = {
			limit = { FROM = { has_country_flag = mtui_invest_50 } }
			change_key = { lhs = Building_AgricultureWealth value = 50 }
			FROM = { add_treasury = -50 }
		}
		if = {
			limit = { FROM = { has_country_flag = mtui_invest_100 } }
			change_key = { lhs = Building_AgricultureWealth value = 100 }
			FROM = { add_treasury = -100 }
		}
		if = {
			limit = { FROM = { has_country_flag = mtui_invest_250 } }
			change_key = { lhs = Building_AgricultureWealth value = 250 }
			FROM = { add_treasury = -250 }
		}
		if = {
			limit = { FROM = { has_country_flag = mtui_invest_500 } }
			change_key = { lhs = Building_AgricultureWealth value = 500 }
			FROM = { add_treasury = -500 }
		}
	}
	tooltip = mtui_bw_invest_agriculture_tt
}

custom_button = {
	name = mtui_bw_select_states
	potential = { owned_by = FROM }
	effect = {
		hidden_effect = {
			if = {
				limit = { has_province_flag = UI_Select }
				owner = { every_owned_province = { limit = { is_state = yes } Pow_UI_R = yes } }
			}
