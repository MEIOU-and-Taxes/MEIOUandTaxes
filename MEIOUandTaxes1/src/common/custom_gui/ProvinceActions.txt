# ROOT is the viewed province; FROM is the player using the interface.
custom_button = {
	name = province_cycle_class_layer_button
	potential = {
		owned_by = FROM
		FROM = { ai = no }
	}
	trigger = {}
	effect = {
		hidden_effect = {
			FROM = { Prov_CycleClassLayer = yes }
		}
	}
	tooltip = province_cycle_class_layer_tooltip
}

custom_button = {
	name = province_select_button
	potential = {
		owned_by = FROM
		FROM = { ai = no }
		NOT = {
			OR = {
				has_province_flag = Pin_Show
				has_province_flag = UI_Select
			}
		}
	}
	trigger = {
		FROM = { NOT = { has_country_flag = UI_Freeze } }
	}
	effect = {
		hidden_effect = { Pow_UI = yes }
	}
	tooltip = province_select_tooltip
}

custom_button = {
	name = province_deselect_button

		}
	}
	trigger = {
		FROM = { NOT = { has_country_flag = UI_Freeze } }

	}
