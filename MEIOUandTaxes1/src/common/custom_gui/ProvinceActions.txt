# ROOT is the viewed province; FROM is the player using the interface.
custom_button = {
	name = province_cycle_class_layer_button
	potential = {
		owned_by = FROM
		FROM = { ai = no }
	}
	trigger = {}
	effect = {
		hidden_effect = {
			FROM = { Prov_CycleClassLayer = yes }
		}
	}
	tooltip = province_cycle_class_layer_tooltip
}

custom_button = {
	name = province_select_button
	potential = {
		owned_by = FROM
		FROM = { ai = no }
	}
	trigger = {
		FROM = { NOT = { has_country_flag = UI_Freeze } }
	}
	effect = {
		hidden_effect = { Prov_ToggleSelection = yes }
	}
	tooltip = province_select_tooltip
}

custom_button = {
	name = province_clear_selection_button
	potential = {
		owned_by = FROM
		FROM = { ai = no }
	}
	trigger = {
		has_global_flag = UI_Select
		OR = {
			NOT = { has_global_flag = multiplayer }
			FROM = {
				any_owned_province = {
					OR = {
						has_province_flag = Pin_Show
						has_province_flag = UI_Select
					}
				}
			}
		}
	}
	effect = {
		hidden_effect = {
			FROM = {
				if = {
					limit = { has_global_flag = multiplayer }
					country_event = { id = SYS_Pin.009 }
				}
				else = {
					country_event = { id = SYS_Pin.008 }
				}
			}
		}
	}
	tooltip = province_clear_selection_tooltip
}
