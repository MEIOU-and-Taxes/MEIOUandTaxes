# Every owned province (scope), PREV the country: adds the province's $key$ to the country's $total$
# (keys of different names add up through Tmp_9)
TopBar_Add = {
	set_key = { lhs = Tmp_9 which = $key$ }
	PREV = {
		set_key = { lhs = Tmp_9 which = PREV }
		change_key = { lhs = $total$ which = Tmp_9 }
	}
}

# Rounds $key$ to a whole number (TopBar_Round1: to one decimal). Keys are fixed point with three decimals:
# x * 0.001 * 1000 drops the fraction, adding 0.5 (or -0.5) first makes it round.
TopBar_Round = {
	if = {
		limit = { check_key = { lhs = $key$ value = 0 } }
		change_key = { lhs = $key$ value = 0.5 }
	}
	else = {
		change_key = { lhs = $key$ value = -0.5 }
	}
	multiply_key = { lhs = $key$ value = 0.001 }
	multiply_key = { lhs = $key$ value = 1000 }
}
TopBar_Round1 = {
	multiply_key = { lhs = $key$ value = 10 }
	TopBar_Round = { key = $key$ }
	divide_key = { lhs = $key$ value = 10 }
}
