# $elite$_$type$ (a class's property of one type) as a percentage of Building_$type$ (all property of that type)
Atlas_PrpShare = {
	if = {
		limit = { check_key = { lhs = Building_$type$ value = 0.001 } }
		set_key = { lhs = $elite$_Prp$type$ which = $elite$_$type$ }
		divide_key = { lhs = $elite$_Prp$type$ which = Building_$type$ }
		multiply_key = { lhs = $elite$_Prp$type$ value = 100 }
	}
	else = { set_key = { lhs = $elite$_Prp$type$ value = 0 } }
}

# $key$ (a copy of one budget line) as a whole-number percentage of $total$
Atlas_Share = {
	if = {
		limit = { check_key = { lhs = $total$ value = 0.001 } }
		divide_key = { lhs = $key$ which = $total$ }
		multiply_key = { lhs = $key$ value = 100 }
		TopBar_Round = { key = $key$ }
	}
	else = { set_key = { lhs = $key$ value = 0 } }
}
