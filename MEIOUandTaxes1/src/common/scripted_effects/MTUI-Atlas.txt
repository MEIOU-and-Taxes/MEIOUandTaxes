# $key$ (a copy of one budget line) as a whole-number percentage of $total$
Atlas_Share = {
	if = {
		limit = { check_key = { lhs = $total$ value = 0.001 } }
		divide_key = { lhs = $key$ which = $total$ }
		multiply_key = { lhs = $key$ value = 100 }
		TopBar_Round = { key = $key$ }
	}
	else = { set_key = { lhs = $key$ value = 0 } }
}

# Every owned province (scope), PREV the country: adds the province's $key$ times its development to the country's
# $total$ (a tenth of it, against overflow, as Nat_SumAdminHelper1 does)
Atlas_AddWeighted = {
	set_key = { lhs = Tmp_9 which = $key$ }
	multiply_key = { lhs = Tmp_9 which = Dev_Total }
	multiply_key = { lhs = Tmp_9 value = 0.1 }
	PREV = {
		set_key = { lhs = Tmp_9 which = PREV }
		change_key = { lhs = $total$ which = Tmp_9 }
	}
}
# $key$ (Atlas_AddWeighted sums) divided by the total development: the development-weighted average
Atlas_Weighted = {
	if = {
		limit = { check_key = { lhs = Dev_Total value = 0.001 } }
		divide_key = { lhs = $key$ which = Dev_Total }
		multiply_key = { lhs = $key$ value = 10 }
	}
	else = { set_key = { lhs = $key$ value = 0 } }
}
