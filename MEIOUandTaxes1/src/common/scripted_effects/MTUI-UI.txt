# Country scope. Refreshes what the top bar and the Nation Atlas show with fresh national sums: Nat_Main ends in
# Nat_Display (TopBar_Refresh, Atlas_Refresh while the Atlas is open); the sums of Nat_SumStats come after it.
mtui_refresh_atlas = {
	Nat_Main = yes
	Nat_SumStats = yes
	TopBar_Refresh = yes
	Atlas_Refresh = yes
}

# Nation Atlas navigation (country scope). The Atlas window itself is shown via custom_window potentials.
mtui_atlas_close = {
	clr_country_flag = mtui_atlas_open
	clr_country_flag = mtui_atlas_tab_overview
	clr_country_flag = mtui_atlas_tab_budget
	clr_country_flag = mtui_atlas_tab_taxes
	clr_country_flag = mtui_atlas_tab_estate_no
	clr_country_flag = mtui_atlas_tab_estate_bg
	clr_country_flag = mtui_atlas_tab_estate_cl
	clr_country_flag = mtui_atlas_tab_estate_tr
	clr_country_flag = mtui_atlas_tab_state
}
mtui_atlas_open_tab = {
	mtui_atlas_close = yes
	set_country_flag = mtui_atlas_open
	set_country_flag = mtui_atlas_tab_$tab$
}
# open a tab with fresh values (top bar shortcuts)
mtui_atlas_open_refresh = {
	mtui_atlas_open_tab = { tab = $tab$ }
	# refresh through a hidden event: the game evaluates the conditions inside a button's effect (province loops)
	# from the country to build its tooltip, which logs scope errors; it does not look into events
	hidden_effect = { country_event = { id = mtui_ui.2 } }
}
mtui_atlas_open_estate_no = { mtui_atlas_open_tab = { tab = estate_no } }
mtui_atlas_open_estate_bg = { mtui_atlas_open_tab = { tab = estate_bg } }
mtui_atlas_open_estate_cl = { mtui_atlas_open_tab = { tab = estate_cl } }
mtui_atlas_open_estate_tr = { mtui_atlas_open_tab = { tab = estate_tr } }
