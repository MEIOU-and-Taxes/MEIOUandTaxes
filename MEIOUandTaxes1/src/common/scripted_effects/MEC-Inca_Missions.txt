inca_conquest_reward = {
	custom_tooltip = inca_conquest_reward_$area$_tt
	hidden_effect = {
		every_owned_province = {
			limit = {
				area = $area$
				NOT = { is_core = ROOT }
			}
			add_province_modifier = {
				name = faster_integration
				duration = 5475
			}
		}
	}
	if = {
		limit = {
			any_subject_country = {
				any_owned_province = { area = $area$ }
			}
		}
		add_country_modifier = {
			name = inca_curaca_reciprocity
			duration = 5475
		}
	}
}


andean_claim_neighbouring_cities = {
	custom_tooltip = andean_claim_neighbouring_cities_tt
	hidden_effect = {
		peru_region = {
			limit = {
				is_city = yes
				NOT = { country_or_non_sovereign_subject_holds = ROOT }
				any_neighbor_province = { owned_by = ROOT }
			}
			add_permanent_claim = ROOT
		}
		bolivia_region = {
			limit = {
				is_city = yes
				NOT = { country_or_non_sovereign_subject_holds = ROOT }
				any_neighbor_province = { owned_by = ROOT }
			}
			add_permanent_claim = ROOT
		}
	}
}

andean_capital_area_reward = {
	custom_tooltip = andean_capital_area_reward_tt
	hidden_effect = {
		capital_scope = { save_event_target_as = andean_capital }
		every_owned_province = {
			limit = {
				area = event_target:andean_capital
				NOT = { is_core = ROOT }
			}
			add_province_modifier = {
				name = faster_integration
				duration = 5475
			}
		}
	}
}
