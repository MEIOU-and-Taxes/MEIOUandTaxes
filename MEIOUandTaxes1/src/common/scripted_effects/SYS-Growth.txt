
# Main function to handle pop birth and death for each province
# Triggered by pulse system and event POP_Growth.001
Growth_GetBirthProv = {
	set_province_flag = POP_ChangedPop # This flags makes the game recalculate a number of pop characteristics after the monthly tick
	
    # Handle Birth and Death for each class with seperate values for birth and death rate
    # There is a exception handling for urban birthing
    Growth_GetBirth = { class = SF rBirth = 1.00 rDeath = 1.05 rural = yes }
    Growth_GetBirth = { class = NM rBirth = 1.00 rDeath = 1.25 rural = yes }
    Growth_GetBirth = { class = RE rBirth = 0.80 rDeath = 1.25 urban = yes }
    Growth_GetBirth = { class = NO rBirth = 1.25 rDeath = 1.05 }
    Growth_GetBirth = { class = BG rBirth = 0.80 rDeath = 1.50 }
    Growth_GetBirth = { class = CL rBirth = 0.80 rDeath = 1.50 clergy = yes }

    # Set how many literates there are
    # Later move Clergy in here as well, when adding clergy privs
    # 1% of resident population
    if = {
        limit = {
            check_key = { lhs = RE_Total value = 0.001 }
        }
        set_key = { lhs = LT_Total which = RE_Total }
		multiply_key = { lhs = LT_Total value = 0.01 }

        set_key = { lhs = LT_Ratio which = LT_Total }
        multiply_key = { lhs = LT_Ratio value = 100 }
        divide_key = { lhs = LT_Ratio which = RE_Total }
    }
    else = { 
        set_key = { lhs = LT_Total value = 0 }
        set_key = { lhs = LT_Ratio value = 0 }
    }

    # Update overall pop birth key for all classes
    # --> the birth keys are the sum of birth and death
    set_key = { lhs = Class_RuralBirth      which = SF_Birth }
    change_key = { lhs = Class_RuralBirth   which = NM_Birth }

    set_key = { lhs = Class_UrbanBirth which = RE_Birth }

    set_key = { lhs = Class_NotablesBirth       which = CL_Birth }
    change_key = { lhs = Class_NotablesBirth    which = BG_Birth }
    change_key = { lhs = Class_NotablesBirth    which = NO_Birth }

    set_key = { lhs = Class_TotalBirth      which = Class_RuralBirth }
    change_key = { lhs = Class_TotalBirth   which = Class_UrbanBirth }
    change_key = { lhs = Class_TotalBirth   which = Class_NotablesBirth }

    set_key = { lhs = Class_RuralChange which = Class_RuralBirth }
    set_key = { lhs = Class_UrbanChange which = Class_UrbanBirth }
    set_key = { lhs = Class_NotablesChange which = Class_NotablesBirth }
    set_key = { lhs = Class_TotalChange which = Class_TotalBirth }
}

# Handle birth and death calculation for each class
# Inputs:
# $class$ = SF | NM | RE | NO | BG | CL
# rBirth = decimal number (multiplier) - Multiplier on birth rate
# rDeath = decimal number (multiplier) - Multiplier on death rate
Growth_GetBirth = {
    if = {
        limit = {
            check_key = { lhs = $class$_Total value = 0.001 } # If there are pops do calc
        }
        ### CASE SWITCH
        # CASE $class$ LifeC Fulfilment >= Growth_PointNeutralFill%
		if = {
			limit = {
				check_key = { lhs = $class$_FillLifeC value = 0.85 } # `Growth_PointNeutralFill~value`
			}
            # Tmp_0 = LifeCFullfilment - Growth_PointNeutralFill
            # Tmp_1 = LifeCFullfilment
			set_key = { lhs = Tmp_0 which = $class$_FillLifeC }
            if = {
                limit = {
                    check_key = { lhs = $class$_FillLifeC value = 2.33 } # `Growth_PointMaxFulfill~value`
                }
                set_key = { lhs = Tmp_0 value = 2.33 } # `Growth_PointMaxFulfill~value`
            }
			set_key = { lhs = Tmp_1 which = Tmp_0 }
			subtract_key = { lhs = Tmp_0 value = 0.85 } # `Growth_PointNeutralFill~value`
			
			# Birth Rate formula:
			#  (LifeCFulfillment - Growth_PointNeutralFill) * Growth_MultPositiveFill
			# (--------------------------------------------------------------------- * (LifeCFulfillment - Growth_PointNeutralFill) + 1) * rBirth
			#                             LifeCFulfillment
			
			set_key = { lhs = Growth_Birth which = Tmp_0 }
			multiply_key = { lhs = Growth_Birth value = 1.0 }   # `Growth_MultPositiveFill~value` ## TOO STRONK
			divide_key = { lhs = Growth_Birth which = Tmp_1 }
			multiply_key = { lhs = Growth_Birth which = Tmp_0 }
			change_key = { lhs = Growth_Birth value = 1 }
			multiply_key = { lhs = Growth_Birth value = $rBirth$ }
			
			## Modifer on pop growth based on usage of land / fisheries in the province
			# The more unused land / fisheries in % the higher the bonus to birth rate due to abundand ressoure and opportunity
            # Applies by the condition rural (currently used for peasants and nomads)
			[[rural]
				set_key = { lhs = Tmp_2 value = 0 }
				set_key = { lhs = Tmp_3 value = 0 }

                ### Calculate multiplier by agricultural filling
				if = {
					limit = {
						has_province_flag = Building_AgricultureHas
						check_key = { lhs = Building_AgricultureFill value = 0.001 }
					}
					set_key = { lhs = Tmp_2 which = Building_AgricultureFill }
                    set_key = { lhs = Tmp_4 which = Tmp_2 }

					change_key = { lhs = Tmp_2 value = 0.30 }   # `Growth_BaseDividendBirthFromEmptyAgriculture~value`
                    multiply_key = { lhs = Tmp_4 value = 1.30 } # `Growth_MultDivisorBirthFromEmptyAgriculture~value`
                    change_key = { lhs = Tmp_4 value = 0.1 }   # `Growth_BaseDivisorBirthFromEmptyAgriculture~value`
					divide_key = { lhs = Tmp_2 which = Tmp_4 }
				}
                ### Calculate multiplier by fishery filling
				if = {
					limit = {
						has_province_flag = Building_FisheryHas
						check_key = { lhs = Building_FisheryFill value = 0.001 }
					}

					set_key = { lhs = Tmp_3 which = Building_FisheryFill }
                    set_key = { lhs = Tmp_4 which = Tmp_3 }

					change_key = { lhs = Tmp_3 value = 0.30 }   # `Growth_BaseDividendBirthFromEmptyFishery~value`
                    multiply_key = { lhs = Tmp_4 value = 1.30 } # `Growth_MultDivisorBirthFromEmptyFishery~value`
                    change_key = { lhs = Tmp_4 value = 0.1 }   # `Growth_BaseDivisorBirthFromEmptyFishery~value`
					divide_key = { lhs = Tmp_3 which = Tmp_4 }
				}

				### Weight Fishery and Agriculture multiplier by some factor
				if = {
					limit = {
						check_key = { lhs = Tmp_2 value = 0.001 }
						check_key = { lhs = Tmp_3 value = 0.001 }
					}
					multiply_key = { lhs = Tmp_2 value = 0.8 }  # `Growth_ScaleBirthFromEmptyAgriculture~value`
					multiply_key = { lhs = Tmp_3 value = 0.2 }  # `Growth_ScaleBirthFromEmptyFishery~value`

					change_key = { lhs = Tmp_2 which = Tmp_3 }
				}

				### Check modifier constraints
				if = {
					limit = {
						NOT = { is_key_equal = { lhs = Tmp_2 value = 0 } }
					}

					if = {
						limit = {
							check_key = { lhs = Tmp_2 value = 1.5 }            # `Growth_MaxBirthFromEmpty~value`
						}
						set_key = { lhs = Tmp_2 value = 1.5 }                  # `Growth_MaxBirthFromEmpty~value`
					}

					else_if = {
						limit = {
							NOT = { check_key = { lhs = Tmp_2 value = 0.95 } }  # `Growth_MinBirthFromEmpty~value`
						}
						set_key = { lhs = Tmp_2 value = 0.95 }                  # `Growth_MinBirthFromEmpty~value`
					}

					multiply_key = { lhs = Growth_Birth which = Tmp_2 }
				}
			]


			set_key = { lhs = Growth_Death value = $rDeath$ }
            
            # Special Handling for Urban death rates
			[[urban]
				if = {
					limit = {
						always = $urban$
					}
					
                    Infra_GetAmenities_Cap = { return = Tmp_0 } # Get cap of amenities after which higher death rates are applied
                    
                    ## If above the cap, recalculate birth rate for this case
					if = {
						limit = {
							check_key = { lhs = RE_Total which = Tmp_0 }
						}

                        # Death rate formula:
                        # (Residents - AmenitiesCap)
                        # -------------------------   * rDeath * Growth_DeathMultFromLowAmenities + rDeath
                        #         Residents
						set_key = { lhs = Growth_Death which = RE_Total }
						subtract_key = { lhs = Growth_Death which = Tmp_0 }
						divide_key = { lhs = Growth_Death which = RE_Total }
						multiply_key = { lhs = Growth_Death value = $rDeath$ }
						multiply_key = { lhs = Growth_Death value = 2.0 } # `Growth_DeathMultFromLowAmenities~value`
						change_key = { lhs = Growth_Death value = $rDeath$ }
					}
				}
			]
		}
		else = { # CASE $class$ LifeC Fulfilment < Growth_PointNeutralFill%
			
            # Birth rate formula: LifeCFulfillment * 0.782 + 0.414
            set_key = { lhs = Growth_Birth which = $class$_FillLifeC }
			multiply_key = { lhs = Growth_Birth value = 0.689 } # `Growth_MultNegativeFill~value`
			change_key = { lhs = Growth_Birth value = 0.414 }   # `Growth_BaseNegativeFill~value`
			
			set_key = { lhs = Growth_Death value = $rDeath$ }

            # Special Handling for Urban death rates
			[[urban]
				if = {
					limit = {
						always = $urban$
					}

					Infra_GetAmenities_Cap = { return = Tmp_0 } # Get cap of amenities after which higher death rates are applied

                    if = {
						limit = {
							check_key = { lhs = RE_Total which = Tmp_0 }
						}
					    
                        # Death rate formula:
                        # (Residents - AmenitiesCap)
                        # -------------------------   * rDeath * Growth_DeathMultFromLowAmenities + rDeath
                        #         Residents
						set_key = { lhs = Growth_Death which = RE_Total }
						subtract_key = { lhs = Growth_Death which = Tmp_0 }
						divide_key = { lhs = Growth_Death which = RE_Total }
						multiply_key = { lhs = Growth_Death value = $rDeath$ }
						multiply_key = { lhs = Growth_Death value = 2.0 } # `Growth_DeathMultFromLowAmenities~value`
						change_key = { lhs = Growth_Death value = $rDeath$ }
					}
				}
			]

			divide_key = { lhs = Growth_Death which = Growth_Birth } # Get birth death rate ratio

			multiply_key = { lhs = Growth_Birth value = $rBirth$ } # Apply hardcoded modifier on birth rate
		}

        # Apply API modifiers to Birth and Death rates
		multiply_key = { lhs = Growth_Birth which = Modi_Birth }
		multiply_key = { lhs = Growth_Birth which = Modi_Birth$class$ }
		multiply_key = { lhs = Growth_Death which = Modi_Death }

        # Subtract Birth and Death Rate
		set_key = { lhs = $class$_Birth which = Growth_Birth }
		subtract_key = { lhs = $class$_Birth which = Growth_Death }

        # Multiply combined birth/death rate with class total to get absolute pop change
		multiply_key = { lhs = $class$_Birth which = $class$_Total }
		multiply_key = { lhs = $class$_Birth value = 0.01 }  # `Growth_GrowthSpeed~value`

		# Clergy above its ideal size
		[[clergy]
			if = {
				limit = {
					check_key = { lhs = Ideal_ClergySize value = 0.001 } # not yet derived -> no excess death
					check_key = { lhs = CL_Total which = Ideal_ClergySize }
				}
				set_key = { lhs = Tmp_0 which = CL_Total }
				subtract_key = { lhs = Tmp_0 which = Ideal_ClergySize }
				multiply_key = { lhs = Tmp_0 value = 0.1 } # share of excess dying per year <- use this for balance
				subtract_key = { lhs = CL_Birth which = Tmp_0 }
			}
		]

        # Apply change to actual total pop key
		change_key = { lhs = $class$_Total which = $class$_Birth }

        # Exception handling for negative pop values
		if = {
			limit = {
				NOT = {
					check_key = { lhs = $class$_Total value = 0 }
				}
			}
			set_key = { lhs = $class$_Total value = 0 }
		}

		set_key = { lhs = Growth_Birth value = 0 }
		set_key = { lhs = Growth_Death value = 0 }
	}
	else = { # Exception handling for no local pop of $class$
		set_key = { lhs = $class$_Birth value = 0 }
		set_key = { lhs = $class$_Total value = 0 }
	}
}

# Main function to handle migration for all provinces and countries
# Triggered by pulse system and event POP_Growth.002
#
# Every class migrates in its own pool: each growth zone has a separate average pressure, outflow and inflow per
# class, so a peasant leaving a province can only arrive as a peasant elsewhere. Migration never changes the class of
# a pop; moving between classes is left to the promotion system.
# Classes taking part: SF, NM, RE, BG (BG at half speed). NO and CL are disabled. To enable one, add its
# Growth_Mig*<class> keys to vars.txt and add its calls next to the BG ones below, in Growth_ZoneReset, Growth_ZoneSum,
# Growth_ZoneLocal, Growth_ZoneDistribute and Growth_ZoneHelper.
Growth_DoMove = {
    # Prepare migration in all valid provinces
    every_province = {
        limit = {
            isValidProv = yes
        }
        ### Calculate migratory pressure for each class based on labor (price) type
        Growth_SetPressure = { class = SF labor = R peasant=yes }
        Growth_SetPressure = { class = NM labor = NM nomad=yes }
        Growth_SetPressure = { class = RE labor = UL urban=yes }
        Growth_SetPressure = { class = BG labor = BG burgher=yes }

		### This effectively disables noble and clergy migration
        #Growth_SetPressure = { class = NO labor = NO }
        #Growth_SetPressure = { class = CL labor = CL }
		set_key = { lhs = Growth_PressureNO value = 0 }
		set_key = { lhs = Growth_PressureCL value = 0 }

		# Tmp_8 = peasant freedom multiplier
		Growth_SetFreedomMult = yes

		# Tmp_<tmp> = possible migrants, Growth_MigPop<class> = pops that count for the zone average
		Growth_SetClassMigrants = { class = SF tmp = 2 freedom = Tmp_8 }
		Growth_SetClassMigrants = { class = NM tmp = 3 }
		Growth_SetClassMigrants = { class = RE tmp = 4 }
		Growth_SetClassMigrants = { class = BG tmp = 6 }

        # Reset existing migration variables of last year
        set_key = { lhs = SF_In value = 0 }
        set_key = { lhs = NM_In value = 0 }
        set_key = { lhs = RE_In value = 0 }
        set_key = { lhs = NO_In value = 0 }
        set_key = { lhs = BG_In value = 0 }
        set_key = { lhs = CL_In value = 0 }

        set_key = { lhs = SF_Out value = 0 }
        set_key = { lhs = NM_Out value = 0 }
        set_key = { lhs = RE_Out value = 0 }
        set_key = { lhs = NO_Out value = 0 }
        set_key = { lhs = BG_Out value = 0 }
        set_key = { lhs = CL_Out value = 0 }
    }

    # Migration zones: each independent country together with its subjects (4 subject levels at most)
    every_country = {
        limit = {
            isValidCountry = yes
			is_free_or_tributary_trigger = yes
        }
        save_event_target_as = GrowthZone

        Growth_ZoneReset = yes

		# Sum up possible migrants and pops per class in the growth zone
        Growth_ZoneEach = { effect = Growth_ZoneSum }

		# Growth_MigAvg<class> = possible migrants of class in growth zone / pops of class in growth zone
		Growth_ZoneSetAverage = { class = SF tmp = 2 }
		Growth_ZoneSetAverage = { class = NM tmp = 3 }
		Growth_ZoneSetAverage = { class = RE tmp = 4 }
		Growth_ZoneSetAverage = { class = BG tmp = 6 }

        # If there are possible migrants of any class calculate
        if = {
            limit = {
				OR = {
					check_key = { lhs = Growth_MigAvgSF value = 0.001 }
					check_key = { lhs = Growth_MigAvgNM value = 0.001 }
					check_key = { lhs = Growth_MigAvgRE value = 0.001 }
					check_key = { lhs = Growth_MigAvgBG value = 0.001 }
				}
            }
            Growth_ZoneEach = { effect = Growth_ZoneLocal }

            # cap actual migration of each class to the smaller of its outward/inward migration
			Growth_ZoneSetFlow = { class = SF tmp = 2 }
			Growth_ZoneSetFlow = { class = NM tmp = 3 }
			Growth_ZoneSetFlow = { class = RE tmp = 4 }
			Growth_ZoneSetFlow = { class = BG tmp = 6 }

            Growth_ZoneEach = { effect = Growth_ZoneDistribute }

			# give what was lost to truncation to the most attractive province, so arrivals = departures
			Growth_ZoneRemainder = { class = SF tmp = 2 }
			Growth_ZoneRemainder = { class = NM tmp = 3 }
			Growth_ZoneRemainder = { class = RE tmp = 4 }
			Growth_ZoneRemainder = { class = BG tmp = 6 }
        }

        Growth_ZoneReset = yes
    }

	#### Migration Zone Mechanics:
	# The below function repeat the migration calculation for specified regional zones, classes, cultures, etc.
	# This is meant to represent a number of historical situations

    #Holy Roman Empire
    Growth_ZoneHelper = { scope=europe HRE=yes SF=yes RE=yes BG=yes }
    #Italian States
    Growth_ZoneHelper = { scope=italy_superregion cultg=latin SF=yes RE=yes BG=yes }
    #Maya
    Growth_ZoneHelper = { scope=central_america_region SF=yes RE=yes NM=yes }
    #Ireland
    Growth_ZoneHelper = { scope=ireland_region SF=yes RE=yes }
    #Manchuria
    Growth_ZoneHelper = { scope=asia flag=Manchu_Zone SF=yes RE=yes NM=yes }
    #Western Steppe
    Growth_ZoneHelper = { scope=steppes_group flag=WestSteppe_Zone NM=yes }
    #Eastern Steppe
    Growth_ZoneHelper = { scope=steppes_group flag=EastSteppe_Zone NM=yes }
    #Arabian Steppe
    Growth_ZoneHelper = { scope=arabian_steppes_group NM=yes }
    #Saharan Steppe
    Growth_ZoneHelper = { scope=saharan_steppes_group NM=yes }
    #Anatolian Nomads
    Growth_ZoneHelper = { scope=anatolia_steppes_group NM=yes }
    if = {
        limit = {
            BYZ = { has_country_flag = BYZ_MigrationIn }
        }
        Growth_ZoneHelper = { scope=bosphorous_strait_group cultg=greek_group SF=yes RE=yes NM=yes }
    }

	# Sum up all changes in total pops, migration etc. per province and set flag that pop was updated
    every_province = {
        limit = {
            isValidProv = yes
        }
		set_province_flag = POP_ChangedPop

        ## Sum up changes per class category
        set_key = { lhs = Class_RuralIn      which = SF_In }
        change_key = { lhs = Class_RuralIn   which = NM_In }

        set_key = { lhs = Class_UrbanIn which = RE_In }

        set_key = { lhs = Class_NotablesIn       which = CL_In }
        change_key = { lhs = Class_NotablesIn    which = BG_In }
        change_key = { lhs = Class_NotablesIn    which = NO_In }

        set_key = { lhs = Class_TotalIn      which = Class_RuralIn }
        change_key = { lhs = Class_TotalIn   which = Class_UrbanIn }
        change_key = { lhs = Class_TotalIn   which = Class_NotablesIn }

        set_key = { lhs = Class_RuralOut      which = SF_Out }
        change_key = { lhs = Class_RuralOut   which = NM_Out }

        set_key = { lhs = Class_UrbanOut which = RE_Out }

        set_key = { lhs = Class_NotablesOut       which = CL_Out }
        change_key = { lhs = Class_NotablesOut    which = BG_Out }
        change_key = { lhs = Class_NotablesOut    which = NO_Out }

        set_key = { lhs = Class_TotalOut      which = Class_RuralOut }
        change_key = { lhs = Class_TotalOut   which = Class_UrbanOut }
        change_key = { lhs = Class_TotalOut   which = Class_NotablesOut }

        ### THOSE KEYS ARE INITIALLY SET IN BIRTH CALCULATION --> THEY SUM UP TOTAL PROVINCE POP CHANGE
        change_key = { lhs = Class_RuralChange which = Class_RuralIn }
        subtract_key = { lhs = Class_RuralChange which = Class_RuralOut }
        change_key = { lhs = Class_UrbanChange which = Class_UrbanIn }
        subtract_key = { lhs = Class_UrbanChange which = Class_UrbanOut }
        change_key = { lhs = Class_NotablesChange which = Class_NotablesIn }
        subtract_key = { lhs = Class_NotablesChange which = Class_NotablesOut }
        change_key = { lhs = Class_TotalChange which = Class_TotalIn }
        subtract_key = { lhs = Class_TotalChange which = Class_TotalOut }

        change_key = { lhs = SF_Total which = SF_In }
        change_key = { lhs = NM_Total which = NM_In }
        change_key = { lhs = RE_Total which = RE_In }
        change_key = { lhs = NO_Total which = NO_In }
        change_key = { lhs = BG_Total which = BG_In }
        change_key = { lhs = CL_Total which = CL_In }

        subtract_key = { lhs = SF_Total which = SF_Out }
        subtract_key = { lhs = NM_Total which = NM_Out }
        subtract_key = { lhs = RE_Total which = RE_Out }
        subtract_key = { lhs = NO_Total which = NO_Out }
        subtract_key = { lhs = BG_Total which = BG_Out }
        subtract_key = { lhs = CL_Total which = CL_Out }

        set_key = { lhs = Tmp_0 value = 0 }
        set_key = { lhs = Tmp_1 value = 0 }
        set_key = { lhs = Tmp_2 value = 0 }
        set_key = { lhs = Tmp_3 value = 0 }
        set_key = { lhs = Tmp_4 value = 0 }
        set_key = { lhs = Tmp_5 value = 0 }
        set_key = { lhs = Tmp_6 value = 0 }
        set_key = { lhs = Tmp_7 value = 0 }
		Growth_ZoneReset = yes
    }
}

# Peasant freedom multiplier on migration, based on peasant freedom and tenancy reform (province scope)
#
# Tenancy Reform Level     Peasant Freedom
# -------------------- *  ----------------- + 0.1
#          4                     100
#
# Constraints: [0.1, 1]
# Output: Tmp_8
Growth_SetFreedomMult = {
	owner = {
	    set_key = { lhs = Tmp_8 which = Rights_NOTen }
	}

	set_key = { lhs = Tmp_8 which = owner }
	divide_key = { lhs = Tmp_8 value = 4 }

	set_key = { lhs = Tmp_9 which = SF_Freedom }

	divide_key = { lhs = Tmp_9 value = 100 } # move peasant freedom on a scale between [0,1]

	multiply_key = { lhs = Tmp_8 which = Tmp_9 } # calculate multiplier

	change_key = { lhs = Tmp_8 value = 0.1 }
	if = {
	    limit = {
	        check_key = { lhs = Tmp_8 value = 1.001 }
	    }
	    set_key = { lhs = Tmp_8 value = 1 }
	}
	else_if = {
	    limit = {
	        NOT = { check_key = { lhs = Tmp_8 value = 0.1 } }
	    }
	    set_key = { lhs = Tmp_8 value = 0.1 }
	}
}

# Possible migrants and migrating pops of one class in this province (province scope)
### Inputs:
# $class$ = SF | NM | RE | BG
# $tmp$ = tmp key number to use for calculation
# $freedom$ = key with the peasant freedom multiplier (optional)
### Outputs:
# Tmp_$tmp$ = $class$_Total * Growth_Pressure$class$ [* freedom]
# Growth_MigPop$class$ = $class$_Total [* freedom]
Growth_SetClassMigrants = {
	set_key = { lhs = Growth_MigPop$class$ which = $class$_Total }
	[[freedom] multiply_key = { lhs = Growth_MigPop$class$ which = $freedom$ } ]
	set_key = { lhs = Tmp_$tmp$ which = Growth_MigPop$class$ }
	multiply_key = { lhs = Tmp_$tmp$ which = Growth_Pressure$class$ }
}

# Clear all per class migration keys of the current scope (zone leader or province)
Growth_ZoneReset = {
	set_key = { lhs = Tmp_2 value = 0 }
	set_key = { lhs = Tmp_3 value = 0 }
	set_key = { lhs = Tmp_4 value = 0 }
	set_key = { lhs = Tmp_6 value = 0 }
	set_key = { lhs = Growth_MigPopSF value = 0 }
	set_key = { lhs = Growth_MigPopNM value = 0 }
	set_key = { lhs = Growth_MigPopRE value = 0 }
	set_key = { lhs = Growth_MigPopBG value = 0 }
	set_key = { lhs = Growth_MigAvgSF value = 0 }
	set_key = { lhs = Growth_MigAvgNM value = 0 }
	set_key = { lhs = Growth_MigAvgRE value = 0 }

# Local migration of the province, summed up per class and zone
Growth_ZoneLocal = {
	# get the average pressure per class of the zone
    set_key = { lhs = Growth_MigAvgSF which = event_target:GrowthZone }
    set_key = { lhs = Growth_MigAvgNM which = event_target:GrowthZone }
    set_key = { lhs = Growth_MigAvgRE which = event_target:GrowthZone }
    set_key = { lhs = Growth_MigAvgBG which = event_target:GrowthZone }

    # Calculate migration on province level
    # how many pops of each class want to go in vs. out
    Growth_DoMove_HandleLocalMigration = { class = SF tmp = 2 seed = 30 freedom = Tmp_8 }
    Growth_DoMove_HandleLocalMigration = { class = NM tmp = 3 seed = 2.5 }
    Growth_DoMove_HandleLocalMigration = { class = RE tmp = 4 seed = 5 industrial = yes }
    Growth_DoMove_HandleLocalMigration = { class = BG tmp = 6 seed = 0.05 }

    # remember the province with the largest inward migration per class
    Growth_ZoneTrackBest = { class = SF zone = event_target:GrowthZone }
    Growth_ZoneTrackBest = { class = NM zone = event_target:GrowthZone }
    Growth_ZoneTrackBest = { class = RE zone = event_target:GrowthZone }
    Growth_ZoneTrackBest = { class = BG zone = event_target:GrowthZone }

    event_target:GrowthZone = { # sum up all values in growth zone
        change_key = { lhs = Growth_MigOutSF which = PREV } # Total outward migration
        change_key = { lhs = Growth_MigOutNM which = PREV }
        change_key = { lhs = Growth_MigOutRE which = PREV }
        change_key = { lhs = Growth_MigOutBG which = PREV }
        change_key = { lhs = Growth_MigInSF which = PREV } # Total inward migration
        change_key = { lhs = Growth_MigInNM which = PREV }
        change_key = { lhs = Growth_MigInRE which = PREV }
        change_key = { lhs = Growth_MigInBG which = PREV }
    }
}

# Average migratory pressure of a class in the zone (zone scope)
### Inputs:
# $class$ = SF | NM | RE | BG
# $tmp$ = tmp key number holding the summed possible migrants of the class
### Outputs:
# Growth_MigAvg$class$ = possible migrants of class in zone / migrating pops of class in zone (0 if there are none)
# Growth_MigOut$class$, Growth_MigIn$class$ = 0, ready to be summed up
# Growth_MigPop$class$ = 0, ready for Growth_ZoneTrackBest
Growth_ZoneSetAverage = {
	if = {
		limit = {
			check_key = { lhs = Growth_MigPop$class$ value = 0.001 }
		}
		set_key = { lhs = Growth_MigAvg$class$ which = Tmp_$tmp$ }
		divide_key = { lhs = Growth_MigAvg$class$ which = Growth_MigPop$class$ }
	}
	else = {
		set_key = { lhs = Growth_MigAvg$class$ value = 0 }
	}
	set_key = { lhs = Growth_MigOut$class$ value = 0 }
	set_key = { lhs = Growth_MigIn$class$ value = 0 }
	set_key = { lhs = Growth_MigPop$class$ value = 0 }
}

# Remember the province with the largest inward migration of a class in the zone (province scope, after
# Growth_DoMove_HandleLocalMigration). Growth_MigPop$class$ is free at this point and holds the running maximum.
### Inputs:
# $class$ = SF | NM | RE | BG
# $zone$ = scope holding the zone's results (event_target:GrowthZone or PREV)
### Outputs:
# $zone$ Growth_MigPop$class$ = largest inward migration of the class so far
# event_target:GrowthBest$class$ = province with the largest inward migration of the class so far
Growth_ZoneTrackBest = {
	if = {
		limit = {
			check_key = { lhs = Growth_MigIn$class$ value = 0.001 }
		}
		set_key = { lhs = Growth_MigPop$class$ which = $zone$ }
		if = {
			limit = {
				check_key = { lhs = Growth_MigIn$class$ which = Growth_MigPop$class$ }
			}
			set_key = { lhs = Growth_MigPop$class$ which = Growth_MigIn$class$ }
			$zone$ = { set_key = { lhs = Growth_MigPop$class$ which = PREV } }
			save_event_target_as = GrowthBest$class$
		}
		set_key = { lhs = Growth_MigPop$class$ value = 0 }
	}
}

# Keys hold 3 decimals, so the migration of each province is truncated and a little less arrives than left
# (or the other way round). Add the difference to the province with the largest inward migration (zone scope).
### Inputs:
# $class$ = SF | NM | RE | BG
# $tmp$ = tmp key number holding the actual migration of the class (Growth_ZoneSetFlow)
# Growth_MigPop$class$ = departures - arrivals of the class (summed up in Growth_DoMove_DistributeMigration)
### Outputs:
# $class$_In of event_target:GrowthBest$class$ increased (or decreased) by the difference
Growth_ZoneRemainder = {
	if = {
		limit = {
			# something moved, so some province had inward migration and set GrowthBest$class$ in this zone
			check_key = { lhs = Tmp_$tmp$ value = 0.001 }
		}
		event_target:GrowthBest$class$ = {
			set_key = { lhs = Growth_MigPop$class$ which = PREV }
			change_key = { lhs = $class$_In which = Growth_MigPop$class$ }
			if = {
				limit = {
					NOT = { check_key = { lhs = $class$_In value = 0 } }
				}
				set_key = { lhs = $class$_In value = 0 }
			}
			set_key = { lhs = Growth_MigPop$class$ value = 0 }
		}
	}
	set_key = { lhs = Growth_MigPop$class$ value = 0 }
}

# Actual migration of a class in the zone = smaller of its total outward and total inward migration (zone scope)
### Inputs:
# $class$ = SF | NM | RE | BG
# $tmp$ = tmp key number to use for calculation
# Growth_MigOut$class$, Growth_MigIn$class$ = total outward / inward migration of the class in the zone (kept)
### Outputs:
# Tmp_$tmp$ = actual migration of the class (0 if there is no outward or no inward migration)
# Growth_MigPop$class$ = 0, ready to sum up departures - arrivals
Growth_ZoneSetFlow = {
	set_key = { lhs = Growth_MigPop$class$ value = 0 }
	if = {
		limit = {
			check_key = { lhs = Growth_MigOut$class$ value = 0.001 }
			check_key = { lhs = Growth_MigIn$class$ value = 0.001 }
		}
		set_key = { lhs = Tmp_$tmp$ which = Growth_MigOut$class$ }
		if = {
			limit = {
				check_key = { lhs = Tmp_$tmp$ which = Growth_MigIn$class$ }
			}
			set_key = { lhs = Tmp_$tmp$ which = Growth_MigIn$class$ }
		}
	}
	else = {
		set_key = { lhs = Tmp_$tmp$ value = 0 }
	}
}

### Inputs:
# $class$ = SF | NM | RE | BG
# $tmp$ = tmp key number to use for calculation
# $seed$ = pops added to the class total for inward migration, About 20% of the class's typical size per province ;
#          a seed far above the class size spreads the inflow evenly over all attractive provinces.
## Keys:
# Growth_Pressure$class$ = Local outward migratory pressure in % of the population
# Growth_MigAvg$class$ = (PossibleMigrants of class in Growth Zone / Pops of class in growth zone)
## Conditionals:
# $zone$ = enables 50% multiplier on amount of possible migration
# $industrial$ = inward migration bonus if has the Industrialisation institution
### Outputs:
# Growth_MigOut$class$ = Outward migration absolute
# Growth_MigIn$class$ = Inward migration absolute
# Tmp_$tmp$ = Outward or inward migration absolute, whichever applies
Growth_DoMove_HandleLocalMigration = {
	set_key = { lhs = Tmp_$tmp$ value = 0 }
	set_key = { lhs = Growth_MigOut$class$ value = 0 }
	set_key = { lhs = Growth_MigIn$class$ value = 0 }
    if = {
        limit = {
            check_key = { lhs = Growth_Pressure$class$ value = 0.001 } # If pressure per $class$ greater 0.001 in this province
            check_key = { lhs = Growth_MigAvg$class$ value = 0.001 } # and there are possible migrants of $class$ in the zone
        }

		# Public_Tmp9 = Migratory pressure of class / (PossibleMigrants of class in Growth Zone / Pops of class in growth zone)
        set_key = { lhs = Public_Tmp9 which = Growth_Pressure$class$ }
        divide_key = { lhs = Public_Tmp9 which = Growth_MigAvg$class$ }

		# Ceiling at factor of 3
        if = {
            limit = {
                check_key = { lhs = Public_Tmp9 value = 3 }
            }
            set_key = { lhs = Public_Tmp9 value = 3 }
        }

        subtract_key = { lhs = Public_Tmp9 value = 1 }

		## Case: Migratory outward pressure of province >= average pressure of class in zone --> outward migration
        if = {
            limit = {
                check_key = { lhs = Growth_Pressure$class$ which = Growth_MigAvg$class$ }
            }

			#       $class$_Total * (Pressure/Average pressure - 1) [[freedom] * Freedom modifier on migration] [[zone] * 0.5]
			#       ------------------------------------------------------------
			#                                80
			set_key = { lhs = Tmp_$tmp$ which = $class$_Total }
			[[zone]multiply_key = { lhs = Tmp_$tmp$ value = 0.5 }]
			[[freedom] multiply_key = { lhs = Tmp_$tmp$ which = $freedom$ }]
			multiply_key = { lhs = Tmp_$tmp$ which = Public_Tmp9 }
			divide_key = { lhs = Tmp_$tmp$ value = 80 }
            if = {
                limit = {
                    OR = {
                        owner = { is_colonial_nation = yes }
                        is_overseas = yes
                    }
                }
                multiply_key = { lhs = Tmp_$tmp$ value = 0.25 } # reduce outward if colonial
            }

			# Store results
            set_key = { lhs = Growth_MigOut$class$ which = Tmp_$tmp$ }
        }
        else = { # Migratory outward pressure of province < average pressure of class in zone --> inward migration

			# Attractiveness mirrors the outward formula: Average pressure/Pressure - 1, ceiling at factor of 3,
			# so very attractive provinces (e.g. from a goldmine) pull far harder than average ones
			set_key = { lhs = Public_Tmp9 which = Growth_MigAvg$class$ }
			divide_key = { lhs = Public_Tmp9 which = Growth_Pressure$class$ }
			subtract_key = { lhs = Public_Tmp9 value = 1 }
			if = {
				limit = {
					check_key = { lhs = Public_Tmp9 value = 3 }
				}
				set_key = { lhs = Public_Tmp9 value = 3 }
			}

			#       ($class$_Total * [[zone] * 0.5] [[freedom] * Freedom modifier on migration] + seed) * (Average pressure/Pressure - 1)
			#       ------------------------------------------------------------
			#                                40
			set_key = { lhs = Tmp_$tmp$ which = $class$_Total }
			[[zone]multiply_key = { lhs = Tmp_$tmp$ value = 0.5 }]
			[[freedom] multiply_key = { lhs = Tmp_$tmp$ which = $freedom$ }]
			change_key = { lhs = Tmp_$tmp$ value = $seed$ }
			multiply_key = { lhs = Tmp_$tmp$ which = Public_Tmp9 }
			divide_key = { lhs = Tmp_$tmp$ value = 40 }
			[[industrial]
			# Increase urban inward migration in lategame
			if = {
				limit = {
					owner = { has_institution = Industrialisation }
				}
				multiply_key = { lhs = Tmp_$tmp$ value = 2 }
			}
			]

			# Store results
			set_key = { lhs = Growth_MigIn$class$ which = Tmp_$tmp$ }
		}
		set_key = { lhs = Public_Tmp9 value = 0 }
    }
}

# This function distributes the migration of a class happening on the provinces inside the growth zone
### Inputs:
# $class$ = SF | NM | RE | BG
# $tmp$ = tmp key number to use for calculation
# $zone$ = scope holding the zone's results (event_target:GrowthZone or PREV)
#
## Keys:
# Growth_MigOut$class$, Growth_MigIn$class$ = Local outward / inward migration absolute, only one of them > 0
# $zone$ Tmp_$tmp$ = actual migration of class in zone
# $zone$ Growth_MigOut$class$, Growth_MigIn$class$ = total outward / inward migration of class in zone
#
### Outputs:
# $class$_In | $class$_Out = Migration for this province (mutually exclusive, either in or out per class)
# $zone$ Growth_MigPop$class$ = departures - arrivals summed up, for Growth_ZoneRemainder
#
#   local migration * 1000
# --------------------------------------------- * actual migration of class in zone / 1000
#  total migration of class in zone (same side)
#
# Keys hold 3 decimals and every step truncates. The share is taken in per mille (at most 1000) so it keeps 6 digits;
# a plain share (at most 1) would lose up to 0.001 / share of the migrants, and multiplying the small local and actual
# migration first would lose up to 0.001 / total in every province. Largest value: 1000 * actual migration.
Growth_DoMove_DistributeMigration = {
	if = {
		limit = {
			check_key = { lhs = Growth_MigOut$class$ value = 0.001 }
		}
		multiply_key = { lhs = Growth_MigOut$class$ value = 1000 }
		divide_key = { lhs = Growth_MigOut$class$ which = $zone$ } # share of zone outward migration in per mille
		set_key = { lhs = Tmp_$tmp$ which = Growth_MigOut$class$ }
		multiply_key = { lhs = Tmp_$tmp$ which = $zone$ } # * actual migration
		divide_key = { lhs = Tmp_$tmp$ value = 1000 }
		change_key = { lhs = $class$_Out which = Tmp_$tmp$ }
		set_key = { lhs = Growth_MigPop$class$ which = Tmp_$tmp$ }
		$zone$ = { change_key = { lhs = Growth_MigPop$class$ which = PREV } }
	}
	else_if = {
		limit = {
			check_key = { lhs = Growth_MigIn$class$ value = 0.001 }
		}
		multiply_key = { lhs = Growth_MigIn$class$ value = 1000 }
		divide_key = { lhs = Growth_MigIn$class$ which = $zone$ } # share of zone inward migration in per mille
		set_key = { lhs = Tmp_$tmp$ which = Growth_MigIn$class$ }
		multiply_key = { lhs = Tmp_$tmp$ which = $zone$ } # * actual migration
		divide_key = { lhs = Tmp_$tmp$ value = 1000 }
		change_key = { lhs = $class$_In which = Tmp_$tmp$ }
		set_key = { lhs = Growth_MigPop$class$ which = Tmp_$tmp$ }
		$zone$ = { subtract_key = { lhs = Growth_MigPop$class$ which = PREV } }
	}
	set_key = { lhs = Growth_MigPop$class$ value = 0 }
}

# This function sets the migratory pressure for each class and labor type in percent
# Inputs:
# $class$ = SF | NM | RE | NO | BG | CL
# $labor$ = R | NM | UL | NO | NO | BG | CL (R = Rural, UL = Urban labor)
# Conditionals:
# peasants | nomads | urban | burgher = yes --> enable special conditions to allow migration
#
# Output: Growth_Pressure$class$ in percent - Constraints [0.1, 10]
# Pressure is only used relative to the zone average, so the scale does not change migration. Percent instead of a
# fraction (0.1) gives it enough decimals: keys only hold 3 decimals, and at a fraction most provinces of a class tied.
Growth_SetPressure = {
    # Generate preasure if pops exist or labor price in province is > 1
    if = {
        limit = {
            OR = {
				check_key = { lhs = $class$_Total value = 0.001 }
				check_key = { lhs = Class_P$labor$ value = 1.001 }
			}
			[[urban] # only allow resident migration if there is a city
			always = $urban$
			has_province_flag = City_Present
			]
			[[burgher] # only allow burgher migration if there is a commerce industry
			always = $burgher$
			has_province_flag = Industry_32Present
			]
			[[peasant] # only allow peasant migration if there are rural industries
			always = $peasant$
			has_province_flag = Rural_Present
			]
			[[nomad] # only allow nomad migration if there is nomad industry
			always = $nomad$
			has_province_flag = Industry_8Present
			]
        }
		
        ### Calculate migration (outward?) pressure per class in %:
        #                    10
        #      ------------------------------
        #        Province Labor Cost per Type
        # 10 = 10% is the maximum pressure it can exert at wage 1.
        set_key = { lhs = Growth_Pressure$class$ value = 10 }
		divide_key = { lhs = Growth_Pressure$class$ which = Class_P$labor$ }

		[[urban]
			if = {
				limit = { 
					always = $urban$
					has_province_flag = City_Present
				}
                ### Apply amenities cap multiplier on migration into cities
                # Constraints: [Growth_MinBaseMigrationPressureEmptyAmenities, Growth_MaxMigrationPressureEmptyAmenities]
                Infra_GetAmenities_Cap = { return = Tmp_0 }

                set_key = { lhs = Tmp_1 which = $class$_Total }
				change_key = { lhs = Tmp_1 value = 0.001 } # In case we have 0 residents, if that is somehow possible)
                divide_key = { lhs = Tmp_1 which = Tmp_0 }

                if = {
                    limit = {
                        check_key = { lhs = Tmp_1 value = 2 }
                    }
                    set_key = { lhs = Tmp_1 value = 1.5 }      # `Growth_MaxMigrationPressureEmptyAmenities~value`
                }
                else_if = {
                    limit = {
                        check_key = { lhs = Tmp_1 value = 1 }
                    }
                    multiply_key = { lhs = Tmp_1 value = 0.25 }
                    change_key = { lhs = Tmp_1 value = 0.75 }
                }
                else = {
                    multiply_key = { lhs = Tmp_1 value = 0.2 }  # `Growth_MinMultMigrationPressureEmptyAmenities~value`
                    change_key = { lhs = Tmp_1 value = 0.8 }    # `Growth_MinBaseMigrationPressureEmptyAmenities~value`
                }

				multiply_key = { lhs = Growth_Pressure$class$ which = Tmp_1 }

                ### Apply multiplier based on city size
                # Min: Growth_BaseDividendMigrationPressureCitySize / Growth_BaseDivisorMigrationPressureCitySize
                # Max: Growth_MultDividendMigrationPressureCitySize / Growth_MultDivisorMigrationPressureCitySize
                set_key = { lhs = Tmp_0 which = $class$_Total }
                set_key = { lhs = Tmp_1 which = $class$_Total }
                multiply_key = { lhs = Tmp_0 value = 2 }        # `Growth_MultDividendMigrationPressureCitySize~value`
                multiply_key = { lhs = Tmp_1 value = 1.25 }     # `Growth_MultDivisorMigrationPressureCitySize~value`
                change_key = { lhs = Tmp_0 value = 60 }         # `Growth_BaseDividendMigrationPressureCitySize~value`
                change_key = { lhs = Tmp_1 value = 120 }        # `Growth_BaseDivisorMigrationPressureCitySize~value`
                divide_key = { lhs = Tmp_0 which = Tmp_1 }

				multiply_key = { lhs = Growth_Pressure$class$ which = Tmp_0 }
			}
		]

        # Apply API effects on pressure
		multiply_key = { lhs = Growth_Pressure$class$ which = Modi_PressureMulti$class$ }
		# Modi_PressureAdd is given as a fraction (0.01 = 1%), convert to percent
		set_key = { lhs = Tmp_0 which = Modi_PressureAdd$class$ }
		multiply_key = { lhs = Tmp_0 value = 100 }
		change_key = { lhs = Growth_Pressure$class$ which = Tmp_0 }
		set_key = { lhs = Tmp_0 value = 0 }

		if = {
			limit = {
				NOT = {
					check_key = { lhs = Growth_Pressure$class$ value = 0.1 }
				}
			}
			set_key = { lhs = Growth_Pressure$class$ value = 0.1 }
		}
		if = {
			limit = {
				check_key = { lhs = Growth_Pressure$class$ value = 10 }
			}
			set_key = { lhs = Growth_Pressure$class$ value = 10 }
		}
    }
    else = {
        set_key = { lhs = Growth_Pressure$class$ value = 10 }
    }
}

# Repeats the migration calculation for a regional zone across borders, at half strength (zone = yes).
# Each enabled class migrates in its own pool, as in Growth_DoMove. The results are summed up on the current scope.
### Inputs:
# $scope$ = region, superregion or province group to iterate over
# $flag$ | $HRE$ | $cultg$ | $cult$ = optional province filters
# $SF$ | $NM$ | $RE$ | $BG$ = yes to let the class migrate in this zone
Growth_ZoneHelper = {
    Growth_ZoneReset = yes

    $scope$ = {
        limit = {
            isValidProv = yes
            [[flag] has_province_flag = $flag$ ]
            [[HRE] is_part_of_hre = $HRE$
            OR = {
                culture_group = low_germanic
                culture_group = high_germanic
                culture_group = low_frankish
                culture_group = west_slavic
            }
            ]
            [[cultg] culture_group = $cultg$ ]
            [[cult] culture = $cult$ ]
        }
        Growth_ZoneReset = yes

        [[SF]
        Growth_SetFreedomMult = yes
        Growth_SetClassMigrants = { class = SF tmp = 2 freedom = Tmp_8 }
        ]
        [[NM] Growth_SetClassMigrants = { class = NM tmp = 3 } ]
        [[RE] Growth_SetClassMigrants = { class = RE tmp = 4 } ]
        [[BG] Growth_SetClassMigrants = { class = BG tmp = 6 } ]

        PREV = {
            change_key = { lhs = Tmp_2 which = PREV }
            change_key = { lhs = Tmp_3 which = PREV }
            change_key = { lhs = Tmp_4 which = PREV }
            change_key = { lhs = Tmp_6 which = PREV }
            change_key = { lhs = Growth_MigPopSF which = PREV }
            change_key = { lhs = Growth_MigPopNM which = PREV }
            change_key = { lhs = Growth_MigPopRE which = PREV }
            change_key = { lhs = Growth_MigPopBG which = PREV }
        }
    }

    [[SF] Growth_ZoneSetAverage = { class = SF tmp = 2 } ]
