CE_Main = {
	# Initialization
	ce_cleanup = yes

	ce_pick_begin = yes # setup of the dispatch selection (ce_set_runner_dispatching_province_land)

	if = {
		limit = {
			is_subject_other_than_tributary_trigger = yes
		}
		overlord = {
			if = {
				limit = {
					is_subject_other_than_tributary_trigger = yes
				}
				overlord = {
					save_event_target_as = tt_highest
				}
			}
			else = {
				save_event_target_as = tt_highest
			}
		}
	}
	else = {
		save_event_target_as = tt_highest
	}

	export_ship_speed_multiplier_effect = { which = tv_ship } # Get hardcoded ship spead increase from techs

	# Add modifier as key to all port provinces
	sea_starts = {
		set_key = { lhs = tv_ship which = PREV }
	}

	# Loop through every non empty province held by a country and their subjects - HRE, Japan and the popes get special runner access for all princes/daimyos/catholics
	# Provinces in the special areas that do not belong to the currently evaluated country can still propagate CE (ie to disconected provinces in the HRE)
	# But different countries' provinces' CE values are not to be changed, so they get back-uped and a bit later restored (recalc gets skipped)
	every_province = {
		limit = {
			isValidProv = yes
			OR = {
				country_or_vassal_holds = event_target:tt_highest
				owner = { overlord = { is_subject_other_than_tributary_trigger = yes is_subject_of = event_target:tt_highest } }
				AND = {
					owner = { is_part_of_hre = yes }     # Allow runners through all of the HRE
					event_target:tt_highest = { is_part_of_hre = yes }
				}
				AND = {
					owner = { daimyo_trigger = yes }     # Allow runners through all of Japan
					event_target:tt_highest = { daimyo_trigger = yes }
				}
				AND = {		# Allow mr pope to always send runners trough all cathiloc owned land
					owner = { religion = catholic }
					event_target:tt_highest = { tag = PAP }
				}
				AND = {		# Allow mr anti pope to always send runners through all avignonist owned land
					owner = { religion = avignonist }
					event_target:tt_highest = { tag = AVI }
				}
			}
		}
		set_key = { lhs = tv_ship which = PREV }

		if = {
			limit = {
				has_port = yes
			}
			PREV = {
				change_key = { lhs = tv_port value = 1 }
			}
		}

		if = {
			limit = {
				owned_by = PREV
			}
			PREV = {
				change_key = { lhs = tv_owned_provinces_in_state_in_progress value = 1 }
			}
		}
		else = {
			set_key = { lhs = pv_ce_save which = pv_ce }
			set_key = { lhs = pv_prev_save which = pv_prev }
			set_key = { lhs = pv_ce_pre_save which = pv_ce_pre }
		}

		set_key = { lhs = pv_ce value = 0 }
		set_key = { lhs = pv_prev value = 0 }
		set_key = { lhs = pv_ce_pre value = 0 }

		ce_get_cost_land = { return = tv_cost_land }
		ce_get_cost_port_embark = { return = tv_cost_port_embark }
		ce_get_cost_port_to = { return = tv_cost_port_to }

		set_province_flag = CE_Cand
		ce_pick_mark = yes
	}
	ce_pick_ready = yes

	# Set runner_dispatching_province
	# The runner_dispatching_province sends out runners to all neighboring provinces and sea zones 
	# a nation has access to. Traveltime is calculated for all possible methods a runner can go to a province
	# (land, sea, river) and the shortest traveltime to that province gets saved in that 
	# province's values. Afterwards, a new dispatching province is chosen, until all provinces have a traveltime.
	# Traveltime is saved as a cumulative value from the nation's capital. 
	# Based on this traveltime, final CE is then calculated.
	# Ancient Passes no longer have a unique travel type, instead are a slightly better Land province.
	capital_scope = {
		set_key = { lhs = tv_tt value = 0 }
		
		if = {
			limit = {
				PREV = {
					check_key = { lhs = tv_owned_provinces_in_state_in_progress value = 1 }
				}
			}
			save_event_target_as = runner_dispatching_province
		}
	}

	# Main loop
	while = {
		limit = {
			event_target:runner_dispatching_province = {
				NOT = { has_province_flag = tf_communication_efficiency_final }
			}
		}

		event_target:runner_dispatching_province = {
			set_province_flag = tf_communication_efficiency_final

			if = {
				limit = {
					owned_by = PREV
				}
				PREV = {
					subtract_key = { lhs = tv_owned_provinces_in_state_in_progress value = 1 }
				}
			}
			if = {
				limit = {
					has_port = yes
				}
				PREV = { 
					subtract_key = { lhs = tv_port value = 1 }
				}
			}

			if = {
				limit = {
					PREV = {
						check_key = { lhs = tv_owned_provinces_in_state_in_progress value = 1 }
					}
				}
				ce_do_river = {
					prev = pv_prev
				}
				ce_do_land = {
					prev = pv_prev
				}

				if = {
					limit = {
						PREV = { 
							check_key = { lhs = tv_port value = 1 }
						}
					}
					ce_do_sea = {
						prev = pv_prev
					}
				}
			}
		}
		if = {
			limit = {
				check_key = { lhs = tv_owned_provinces_in_state_in_progress value = 1 }
			}
			ce_set_runner_dispatching_province_land = {
				max = 1800
			}
			
			if = {
				limit = {
					check_key = { lhs = tv_port value = 1 }
				}
				ce_set_runner_dispatching_province_sea = {
					max = 1800
				}
			}
		}
	}
	
	every_province = {
		limit = {
			has_province_flag = tf_river
		}
		clr_province_flag = tf_river
	}
	
	# Evaluation of regional capitols begins here.
	clr_country_flag = ce_capitols 

	# First, we check if there even exist any regional capitols the nation's runners have visited during
	# their initial run from the capital. 
	# In special access zones (HRE, Japan, cahtolic land if popes),
	# other nation's capital capitol's count as a regional.
	if = {
		limit = {
			any_province = {
				has_province_flag = tf_communication_efficiency_final
				
				check_key = { lhs = Infra_Capitol value = 1 }
				
				NOT = {
					is_capital_of = event_target:tt_highest
				}
			}
		}
		set_country_flag = ce_capitols
		
		# We backup the original traveltime from the capital before any reductions are applied for later
		every_province = {
			limit = {
				has_province_flag = tf_communication_efficiency_final
			}
			set_province_flag = tf_communication_efficiency_final_save
			set_key = { lhs = tv_tt_save which = tv_tt }
		}
		
		sea_starts = {
			limit = {
				has_province_flag = tf_communication_efficiency_final
			}
			set_province_flag = tf_communication_efficiency_final_save
			set_key = { lhs = tv_tt_save which = tv_tt }
		}

		# All capitols a nation's runners can use as a regional get flagged
		every_province = {
			limit = {
				has_province_flag = tf_communication_efficiency_final_save	
				
				NOT = {
					is_capital_of = event_target:tt_highest
				}	
			}
			if = {
				limit = {
					check_key = { lhs = Infra_Capitol value = 1 }
				}
				set_province_flag = is_regional_capital
				
				set_key = { lhs = Tmp_9 value = 0 }		# To enforce chaining restriction
			}
	    }

		# All regionals now get evaluated in order by distance to the capital. Flags that mark provinces
		# as already evaluated get cleared for each new regional, and then runners from that regional 
		# spread out until no new dispatch province is found. 
		# This happens if the traveltime to adjacent provinces from all available dispatch provinces
		# is longer than the original traveltime from the capital. 
		every_province = {
			limit = {
				has_province_flag = is_regional_capital
			}

			# Reset flags for done provinces.
			every_province = {
				limit = {
					has_province_flag = tf_communication_efficiency_final_save						
				}
				clr_province_flag = tf_communication_efficiency_final
				clr_province_flag = tf_river
				clr_province_flag = ce_dispatch_cand
			}
			
			sea_starts = {
				limit = {
					has_province_flag = tf_communication_efficiency_final_save
				}
				clr_province_flag = tf_communication_efficiency_final
				clr_province_flag = ce_dispatch_cand
			}	
			
			# Find unusued regional with best distance
			save_event_target_as = runner_dispatching_province
			
			every_province = {
				limit = {
					has_province_flag = is_regional_capital
	
					NOT = {
						has_province_flag = regional_capital_ce_final
					}
					
					NOT = {
						check_key = { lhs = tv_tt which = event_target:runner_dispatching_province }
					}
				}
				save_event_target_as = runner_dispatching_province
			}


			# Apply reduction to traveltime based on capitol level.
			event_target:runner_dispatching_province = {
				if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 1 } } multiply_key = { lhs = tv_tt value = 0.92 } subtract_key = { lhs = tv_tt value = 20 } }
				else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 2 } } multiply_key = { lhs = tv_tt value = 0.85 } subtract_key = { lhs = tv_tt value = 40 } }
				else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 3 } } multiply_key = { lhs = tv_tt value = 0.79 } subtract_key = { lhs = tv_tt value = 60 } }
				else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 4 } } multiply_key = { lhs = tv_tt value = 0.74 } subtract_key = { lhs = tv_tt value = 70 } }
				else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 5 } } multiply_key = { lhs = tv_tt value = 0.69 } subtract_key = { lhs = tv_tt value = 80 } }
				else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 6 } } multiply_key = { lhs = tv_tt value = 0.64 } subtract_key = { lhs = tv_tt value = 90 } }
				else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 7 } } multiply_key = { lhs = tv_tt value = 0.6 } subtract_key = { lhs = tv_tt value = 100 } }
				
				if = {
					limit = {
						NOT = {
							check_key = { lhs = tv_tt value = 0.001 }
						}
					}
					set_key = { lhs = tv_tt value = 0.001 }
				}

				# Safeguard against exploiting chaining of capitols.
				# If the runner origin of a capitol is another capitol,
				# then traveltime cannot be lower than the traveltime of it's runner origin.
				# there can be an exception that was not previously considered
				# If a given province was visited by more than one regional, then there could be a case where,
				# regional A gives better distance to the current regional, but regional B has a better distance itself.
				# In such a case, if capitol is high enough in the current regional, then it can actually be more beneficial for the current regional
				# to have received the runner from regional B, even though before the discount is considered it would give a worse distance
				# If one is to implement such a feature the logic would get pretty more complex, with quite the impact on performance
				# So, it is assumed that such an exception is to be handled by checking only the regional that sent the best runner
				# which regional to receive runner from
				if = {
					limit = {
						check_key = { lhs = Tmp_9 value = 0.001 }
						
						NOT = {
							check_key = { lhs = tv_tt which = Tmp_9 }
						}
					}
					set_key = { lhs = tv_tt which = Tmp_9 }
				}
					
				
				set_province_flag = regional_capital_ce_final		
				set_key = { lhs = Tmp_9 which = tv_tt }
										
				# Main loop for current regional
				while = {
					limit = {
						event_target:runner_dispatching_province = {
							NOT = { has_province_flag = tf_communication_efficiency_final }
						}
					}

					event_target:runner_dispatching_province = {
						set_province_flag = tf_communication_efficiency_final
						
						if = {
							limit = {
								NOT = {
									has_province_flag = is_regional_capital
	
									NOT = {
										has_province_flag = regional_capital_ce_final
									}
								}
							}
							set_key = { lhs = Tmp_9 which = PREV }
						}

						ce_do_river = {
							prev = pv_prev
						}
						ce_do_land = {
							prev = pv_prev
						}
						ce_do_sea = {
							prev = pv_prev
						}
	
		
						ce_set_runner_dispatching_province_land = {
							max = 960
						}
						ce_set_runner_dispatching_province_sea = {
							max = 960
						}
					}
				}
			}
		}

		every_province = {
			limit = {
				has_province_flag = tf_communication_efficiency_final_save
			}
			clr_province_flag = tf_communication_efficiency_final_save
			set_province_flag = tf_communication_efficiency_final
		}
		
		sea_starts = {
			limit = {
				has_province_flag = tf_communication_efficiency_final_save
			}
			clr_province_flag = tf_communication_efficiency_final_save
			set_province_flag = tf_communication_efficiency_final
		}		
		
		every_province = {
			limit = {
				has_province_flag = tf_river
			}
			clr_province_flag = tf_river
		}
	}
	else_if = {
		limit = {
			capital_scope = { check_key = { lhs = Infra_Capitol value = 1 } }
		}
		# No regionals, but a capitol in the capital: backup the unreduced traveltime as well,
		# it is the base of the capitol contribution (CE_Diff) in the CE tooltip.
		every_province = {
			limit = {
				has_province_flag = tf_communication_efficiency_final
			}
			set_key = { lhs = tv_tt_save which = tv_tt }
		}
	}


	# After the regionals have been done, a global traveltime reduction to all provinces the runners
	# passed through based on the capitol level in the nation's capital is applied, 
	# including sea zones and foreign provinces the runners have access to. 
	set_key = { lhs = Tmp_0 value = 1 }

	capital_scope = {
		if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 1 } } PREV = { set_key = { lhs = Tmp_0 value = 0.94 } } }
		else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 2 } } PREV = { set_key = { lhs = Tmp_0 value = 0.88 } } }
		else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 3 } } PREV = { set_key = { lhs = Tmp_0 value = 0.82 } } }
		else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 4 } } PREV = { set_key = { lhs = Tmp_0 value = 0.77 } } }
		else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 5 } } PREV = { set_key = { lhs = Tmp_0 value = 0.73 } } }
		else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 6 } } PREV = { set_key = { lhs = Tmp_0 value = 0.69 } } }
		else_if = { limit = { is_key_equal = { lhs = Infra_Capitol value = 7 } } PREV = { set_key = { lhs = Tmp_0 value = 0.65 } } }
	}

	every_province = {
		limit = {
			check_key = { lhs = tv_tt value = 0.001 }
			has_province_flag = tf_communication_efficiency_final
		}
		set_key = { lhs = Tmp_0 which = PREV }

		multiply_key = { lhs = tv_tt which = Tmp_0 }

		set_key = { lhs = Tmp_0 value = 0 }
	}

	sea_starts = {
		limit = {
			check_key = { lhs = tv_tt value = 0.001 }
			has_province_flag = tf_communication_efficiency_final
		}
		set_key = { lhs = Tmp_0 which = PREV }

		multiply_key = { lhs = tv_tt which = Tmp_0 }

		set_key = { lhs = Tmp_0 value = 0 }
	}	


	# Japan has additional reductions 
	if = {
		limit = {
			has_country_flag = limiting_shugo_power
		}
		every_province = {
			limit = {
				owner = {
					culture_group = japanese_g
					
					OR = {
						is_subject_of_type = daimyo_vassal
						is_subject_of_type = vassal
					}
					
					overlord = {
						has_country_flag = limiting_shugo_power
					}
				}
			}
			set_key = { lhs = Tmp_0 which = PREV }

			multiply_key = { lhs = tv_tt which = Tmp_0 }

			set_key = { lhs = Tmp_0 value = 0 }
			
			set_key = { lhs = japan_overlord_ce which = tv_tt }
			change_key = { lhs = japan_overlord_ce value = 0.001 }
		}
	}
	else_if = {
		limit = {
			culture_group = japanese_g
					
			OR = {
				is_subject_of_type = daimyo_vassal
				is_subject_of_type = decentralized_vassal
				is_subject_of_type = vassal
			}
					
			overlord = {
				has_country_flag = limiting_shugo_power
			}
		}
		every_owned_province = {
			limit = {
				check_key = { lhs = tv_tt value = 0.001 }
				check_key = { lhs = japan_overlord_ce value = 0.001 }
				NOT = {
					check_key = { lhs = japan_overlord_ce value = 1000 }
				}
			}
			change_key = { lhs = tv_tt value = 1000 }
			subtract_key = { lhs = tv_tt which = japan_overlord_ce }
		}
	}

	set_key = { lhs = Tmp_0 value = 0 }

	# Finalize
	ce_finalize_effect = {
		scope = every_owned_province
		ce_var = tv_tt
		time = pv_ce
		clean = yes
	}
	if = {
		limit = {
			OR  = {
				has_country_flag = ce_capitols
				capital_scope = { check_key = { lhs = Infra_Capitol value = 1 } }
			}
		}
		# Finalize
		ce_finalize_effect = {
			scope = every_owned_province
			ce_var = tv_tt_save
			time = pv_ce_pre
			clean = yes
		}
	}

	# If the province belongs to the current country evaluated, set autonomy modifiers. 
	# If not, restore previously saved CE values, so we don't override another nations traveltimes for their runners.
	every_province = {
		limit = {
			OR = {
				country_or_vassal_holds = event_target:tt_highest
				owner = { overlord = { is_subject_other_than_tributary_trigger = yes is_subject_of = event_target:tt_highest } }
				AND = {
					owner = { is_part_of_hre = yes }
					event_target:tt_highest = { is_part_of_hre = yes }
				}
				AND = {
					owner = { daimyo_trigger = yes }
					event_target:tt_highest = { daimyo_trigger = yes }
				}
				AND = {	
					owner = { religion = catholic }
					event_target:tt_highest = { tag = PAP }
				}
				AND = {		
					owner = { religion = avignonist }
					event_target:tt_highest = { tag = AVI }
				}
			}
			isValidProv = yes
		}

		if = {
			limit = {
				NOT = {
					owned_by = PREV
				}
			}
			set_key = { lhs = pv_ce which = pv_ce_save }
			set_key = { lhs = pv_prev which = pv_prev_save }
			set_key = { lhs = pv_ce_pre which = pv_ce_pre_save }
		}
		else = {
			set_key = { lhs = Tmp_1 which = pv_ce }
			
			add_CE_modifier = yes
			
			multiply_key = { lhs = Tmp_1 value = 0.014 } #Corlis: This scales a Value of CE from 0 to 100 to Autonomy reduction. So 100-> 1 Autonomy reduction
			
			set_key = { lhs = pv_CEauto which = Tmp_1 }
			
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 1.500 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 1.000 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 1.024 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.512 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.256 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.128 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.064 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.032 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.016 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.008 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.004 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.002 }
			POP_SetMod = { varname = Tmp_1 modname = Stat_CE type = province value = 0.001 }
			
			set_key = { lhs = Tmp_1 value = 0 }
		}

		# Calc difference in CE
		# This is the difference in the original, unreduced traveltime to a province from the capital
		# vs the reduced traveltime from the global reduction from the capital capitol.
		# and possibly the reduction from regionals. 		
		if = {
			limit = {
				owned_by = PREV
				OR = {
					PREV = { has_country_flag = ce_capitols }
					PREV = { capital_scope = { check_key = { lhs = Infra_Capitol value = 1 } } }
				}	
			}
			set_key = { lhs = CE_Diff which = pv_ce }
			subtract_key = { lhs = CE_Diff which = pv_ce_pre }
		}
		else_if = {
			limit = {
				owned_by = PREV
			}
			set_key = { lhs = CE_Diff value = 0 }
		}

		set_key = { lhs = pv_ce_save value = 0 }
		set_key = { lhs = pv_prev_save value = 0 }
		set_key = { lhs = pv_ce_pre_save value = 0 }
	}
	clr_country_flag = ce_capitols 

	# Clear all temporary variables.
	ce_cleanup = yes
}

compare_and_set_travel_time = {
	if = {
		limit = {
			check_key = { lhs = tv_tt value = 0.001 }
		}
		event_target:runner_dispatching_province = {
			set_key = { lhs = tv_tt_new which = tv_tt }
		}

		set_key = { lhs = tv_tt_new which = event_target:runner_dispatching_province }
		change_key = { lhs = tv_tt_new which = $value$ }

		if = {
			limit = {
				NOT = {
					check_key = { lhs = tv_tt_new which = tv_tt }
				}
			}
			clr_province_flag = tf_river
			set_province_flag = ce_dispatch_cand
			
			set_key = { lhs = tv_tt which = tv_tt_new }

			event_target:runner_dispatching_province = {
				if = {
					limit = {
						is_sea = yes
					}
					set_key = { lhs = tv_id which = $prev$ }
				}
				else = {
					set_key = { lhs = tv_id which = ID_Prov }
				}
			}

			set_key = { lhs = tv_id which = event_target:runner_dispatching_province }

			set_key = { lhs = $prev$ which = tv_id }
		}
	}
	else = {
		clr_province_flag = tf_river
		set_province_flag = ce_dispatch_cand
		
		event_target:runner_dispatching_province = {
			if = {
				limit = {
					is_sea = yes
				}
				set_key = { lhs = tv_id which = $prev$ }
			}
			else = {
				set_key = { lhs = tv_id which = ID_Prov }
			}
		}

		set_key = { lhs = tv_tt which = event_target:runner_dispatching_province }
		set_key = { lhs = tv_id which = event_target:runner_dispatching_province }

		change_key = { lhs = tv_tt which = $value$ }
		set_key = { lhs = $prev$ which = tv_id }
	}
}

ce_finalize_effect = {
	$scope$ = {
		if = {
			limit = {
				has_province_flag = tf_communication_efficiency_final
			}
			set_key = { lhs = $time$ value = 960 }
			subtract_key = { lhs = $time$ which = $ce_var$ }

			if = {
				limit = {
					NOT = {
						check_key = { lhs = $time$ value = 0.001 }
					}
				}
				set_key = { lhs = $time$ value = 0 }
			}
			else = {
				divide_key = { lhs = $time$ value = 9.6 }
			}
		}
		else_if = {
			limit = {
				always = $clean$
			}
			set_key = { lhs = $time$ value = 0 }
		}
	}
}

# Picks the candidate with the lowest travel time. Only the provinces that can be candidates are walked (see CE_Main);
# ties go to the lower province id (ID_Prov), so the pick does not depend on the walking order.
# If nothing qualifies, the dispatching province stays the current one (already final), which ends the caller's loop.
ce_set_runner_dispatching_province_land = {
	event_target:runner_dispatching_province = { save_event_target_as = runner_dispatching_province_cand }
	set_key = { lhs = Tmp_0 which = tv_tt }
	set_key = { lhs = tv_tt value = 1800 }
	set_key = { lhs = CE_BestId value = 1000000 }
	save_event_target_as = ce_acc
	if = {
		limit = { has_global_flag = ce_tree_mode }
		event_target:tt_highest = { ce_each_tree = { effect = ce_pick_land_$max$ } }
	}
	else = {
		every_country = {
			limit = { has_country_flag = ce_area_owner }
			every_owned_province = { ce_pick_land_$max$ = yes }
		}
	}
	event_target:runner_dispatching_province_cand = { save_event_target_as = runner_dispatching_province }
	set_key = { lhs = tv_tt which = Tmp_0 }
	set_key = { lhs = Tmp_0 value = 0 }
	set_key = { lhs = CE_BestId value = 0 }
}
# Setup for the selection; every route finder that uses it (CE_Main, AI_StateBuilding_CE) calls these three:
# ce_pick_begin (country scope) before flagging CE_Cand, ce_pick_mark in every province it flags CE_Cand,
# ce_pick_ready (country scope, after event_target:tt_highest is set) afterwards.
ce_pick_begin = {
	save_event_target_as = ce_country
	every_country = {
		limit = { has_country_flag = ce_area_owner }
		clr_country_flag = ce_area_owner
	}
	set_key = { lhs = CE_CandCount value = 0 }
}
ce_pick_mark = {
	event_target:ce_country = { change_key = { lhs = CE_CandCount value = 1 } }
	owner = { set_country_flag = ce_area_owner }
}
# Which provinces the selection walks: the subject tree of tt_highest if it holds every candidate,
# otherwise (special access: HRE, Japan, the popes) the provinces of every country that owns a candidate
ce_pick_ready = {
	set_key = { lhs = CE_TreeCount value = 0 }
	event_target:tt_highest = { ce_each_tree = { effect = ce_count_tree } }
	if = {
		limit = { is_key_equal = { lhs = CE_TreeCount which = CE_CandCount } }
		set_global_flag = ce_tree_mode
	}
	else = {
		clr_global_flag = ce_tree_mode
	}
}
# one wrapper per max value used by a caller (CE_Main 1800 and 960, AI_StateBuilding_CE 1440)
ce_pick_land_1800 = { ce_pick_land = { max = 1800 } }
ce_pick_land_1440 = { ce_pick_land = { max = 1440 } }
ce_pick_land_960 = { ce_pick_land = { max = 960 } }
ce_pick_land = { # province scope; event_target:ce_acc holds the best travel time (tv_tt) and its province id (CE_BestId)
	if = {
		limit = {
			has_province_flag = ce_dispatch_cand
			NOT = { has_province_flag = tf_communication_efficiency_final }
			has_province_flag = CE_Cand
			NOT = { check_key = { lhs = tv_tt value = $max$ } }
		}
		set_key = { lhs = CE_BestId which = event_target:ce_acc }
		if = {
			limit = {
				OR = {
					NOT = { check_key = { lhs = tv_tt which = event_target:ce_acc } } # lower travel time
					AND = {
						event_target:ce_acc = { check_key = { lhs = tv_tt which = PREV } } # equal travel time ...
						NOT = { check_key = { lhs = ID_Prov which = CE_BestId } }       # ... and lower id
					}
				}
			}
			set_key = { lhs = CE_BestId which = ID_Prov }
			event_target:ce_acc = {
				set_key = { lhs = tv_tt which = PREV }
				set_key = { lhs = CE_BestId which = PREV }
			}
			save_event_target_as = runner_dispatching_province_cand
		}
		set_key = { lhs = CE_BestId value = 0 }
	}
}
# Runs $effect$ in every province of the current country and its subjects (3 levels deep)
ce_each_tree = {
	every_owned_province = { $effect$ = yes }
	every_subject_country = {
		every_owned_province = { $effect$ = yes }
		every_subject_country = {
			every_owned_province = { $effect$ = yes }
			every_subject_country = {
				every_owned_province = { $effect$ = yes }
			}
		}
	}
}
ce_count_tree = {
	if = {
		limit = { has_province_flag = CE_Cand }
		event_target:ce_country = { change_key = { lhs = CE_TreeCount value = 1 } }
	}
}
# Runs right after ce_set_runner_dispatching_province_land. A sea zone replaces the land candidate only if its travel
# time is strictly lower, so the next dispatch is always the closest unsettled province or sea zone (Dijkstra order).
# The land candidate is new (not final) only if the land pick found one.

		}
	}

	set_key = { lhs = tv_tt which = Tmp_0 }
	set_key = { lhs = Tmp_0 value = 0 }
}

# River runners: from the dispatching province, shortest travel along connected river provinces (Dijkstra along the
# river). tf_river_done = river distance (tv_ts) final; tf_river_cand = reached, tv_ts may still shrink. The next
# province to expand is the reached one with the lowest tv_ts (ties: lowest province id), so the result does not
# depend on any walking order.
ce_do_river = {
	if = {
		limit = {
			check_key = { lhs = TN_River value = 1 }
			
			NOT = {
				has_province_flag = tf_river
			}
		}
		save_event_target_as = Prov
		
		set_key = { lhs = tv_ts which = tv_tt }
		change_key = { lhs = tv_ts which = tv_cost_port_embark }
		
		set_global_flag = tf_river
		set_province_flag = tf_river_done
		
		while = {
			limit = {
				has_global_flag = tf_river
			}
			clr_global_flag = tf_river
			
			event_target:Prov = {
				every_neighbor_province = {
					limit = {
						has_province_flag = CE_Cand
						check_key = { lhs = TN_River value = 1 }
						
						NOT = { 
							has_province_flag = tf_river_done 
						}
				
						OR = {
							is_key_equal = { lhs = TN_River which = PREV }
							
							has_province_flag = TN_RiverJoint
							
							PREV = { has_province_flag = TN_RiverJoint }
						}
					}
					export_province_distance_river_province_effect = {
						which = tv_province_distance
						who = event_target:Prov
					}

					if = {
						limit = { has_province_flag = tf_river_cand }
						set_key = { lhs = tv_tt_new which = tv_ts } # river distance found so far
					}
					set_key = { lhs = tv_ts which = PREV }
					change_key = { lhs = tv_ts which = tv_province_distance }
					if = {
						limit = {
							has_province_flag = tf_river_cand
							check_key = { lhs = tv_ts which = tv_tt_new }
						}
						set_key = { lhs = tv_ts which = tv_tt_new } # the earlier one is not longer: keep it
					}
					set_province_flag = tf_river_cand

					set_key = { lhs = tv_tt_new which = tv_ts }
					change_key = { lhs = tv_tt_new which = tv_cost_port_embark }
					
					if = {
						limit = {
							OR = {
								AND = {
									is_key_equal = { lhs = tv_tt value = 0 }
									
									is_capital = no
								}
								
								NOT = {
									check_key = { lhs = tv_tt_new which = tv_tt }
								}
							}
						}
						set_province_flag = tf_river
						set_province_flag = ce_dispatch_cand
						
						set_key = { lhs = tv_tt which = tv_tt_new }
						
						event_target:runner_dispatching_province = {
							set_key = { lhs = tv_id which = ID_Prov }
						}

						set_key = { lhs = tv_id which = event_target:runner_dispatching_province }
						set_key = { lhs = $prev$ which = tv_id }
					}
				}
			}
			
			# next: the reached province with the lowest river distance (event_target:ce_acc, our country, holds the best so far)
			event_target:ce_country = { save_event_target_as = ce_acc }
			event_target:ce_acc = {
				set_key = { lhs = tv_ts value = 1000000 }
				set_key = { lhs = CE_BestId value = 1000000 }
			}
			every_province = {
				limit = {
					has_province_flag = tf_river_cand
				}
				ce_pick_river = yes
			}
			if = {
				limit = {
					event_target:ce_acc = { NOT = { check_key = { lhs = CE_BestId value = 1000000 } } }
				}
				event_target:ce_river_next = {
					clr_province_flag = tf_river_cand
					set_province_flag = tf_river_done
					save_event_target_as = Prov
				}
				set_global_flag = tf_river
			}
			event_target:ce_acc = {
				set_key = { lhs = tv_ts value = 0 }
				set_key = { lhs = CE_BestId value = 0 }
			}
		}
		
		every_province = {
			limit = {
				has_province_flag = tf_river_done
			}
			clr_province_flag = tf_river_done
			
			set_key = { lhs = tv_ts value = 0 }
		}
	}
}
ce_pick_river = { # province scope; event_target:ce_acc holds the best river distance (tv_ts) and its province id (CE_BestId)
	set_key = { lhs = CE_BestId which = event_target:ce_acc }
	if = {
		limit = {
			OR = {
				NOT = { check_key = { lhs = tv_ts which = event_target:ce_acc } } # shorter river distance
				AND = {
					event_target:ce_acc = { check_key = { lhs = tv_ts which = PREV } } # equal distance ...
					NOT = { check_key = { lhs = ID_Prov which = CE_BestId } }       # ... and lower id
				}
			}
		}
		set_key = { lhs = CE_BestId which = ID_Prov }
		event_target:ce_acc = {
			set_key = { lhs = tv_ts which = PREV }
			set_key = { lhs = CE_BestId which = PREV }
		}
		save_event_target_as = ce_river_next
	}
	set_key = { lhs = CE_BestId value = 0 }
}
