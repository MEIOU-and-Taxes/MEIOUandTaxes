namespace = mtui_ui

# Refreshes the top bar and Nation Atlas values. Buttons fire it instead of calling mtui_refresh_atlas: the game
# builds a button's tooltip from the conditions inside its effect (province loops too) and logs scope errors.
country_event = {
	id = mtui_ui.2
	title = no_localisation
	desc = no_localisation
	picture = MEIOU_AND_TAXES_eventPicture
	hidden = yes
	is_triggered_only = yes
	immediate = {
		hidden_effect = { mtui_refresh_atlas = yes }
	}
	option = { name = no_localisation }
}

# Repeating Atlas actions: monthly check while any repeat toggle is on.
country_event = {
	id = mtui_ui.10
	title = no_localisation
	desc = no_localisation
	picture = MEIOU_AND_TAXES_eventPicture
	hidden = yes
	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					has_country_flag = mtui_repeat_tax
					BU_Action_TaxReliefAllowed = yes
				}
				BU_Action_TaxRelief = yes log ="SYS-Politics:[GetYear]:[This.GetName]:Tax Relief:Commoners:+:[This.GovernmentName]"
				mtui_after_action = yes
			}
			if = {
				limit = {
					has_country_flag = mtui_repeat_grain
					OR = {
						NOT = { has_country_flag = mtui_grain_done }
						had_country_flag = { flag = mtui_grain_done days = 1095 }
					}
				}
				if = { limit = { has_country_flag = mtui_grain_last_Full } BU_Action_GrainFull = yes log ="SYS-Politics:[GetYear]:[This.GetName]:Grain Dole Full:Commoners:+:[This.GovernmentName]" mtui_grain_mark = { lvl = Full } }
				else_if = { limit = { has_country_flag = mtui_grain_last_Major } BU_Action_GrainMajor = yes log ="SYS-Politics:[GetYear]:[This.GetName]:Grain Dole Major:Commoners:+:[This.GovernmentName]" mtui_grain_mark = { lvl = Major } }
				else = { BU_Action_GrainMinor = yes log ="SYS-Politics:[GetYear]:[This.GetName]:Grain Dole Minor:Commoners:+:[This.GovernmentName]" mtui_grain_mark = { lvl = Minor } }
				mtui_after_action = yes
			}
			if = {
				limit = {
					OR = {
						has_country_flag = mtui_repeat_tax
						has_country_flag = mtui_repeat_grain
					}
				}
				country_event = { id = mtui_ui.10 days = 30 }
			}
			else = {
				clr_country_flag = mtui_chain
			}
		}
	}
	option = { name = no_localisation }
}
