# America smallpox chain (see America_ContactDisease in SYS-PlagueSpawner.txt)
# Central America is hit 7-8 years after first contact in Central America / the Caribbean,
# or in West America / the Pacific side of Central America (seeds the Pacific coast instead),
# South America follows 5-10 years after Central. North America is triggered by colonies there.
country_event = { #central america smallpox
	id = Plague_Spawner.222
	title = no_localisation
	desc = no_localisation
    picture = no_picture
	fire_only_once = yes
	is_triggered_only = yes
	hidden = yes
	trigger = {
        tag = AAA
        has_global_flag = not_camerica_smallpox
    }
    
    immediate = { #iiiincoooomiiing
        # One seed per zone so the outbreak spreads around the region
        if = { # first contact came from the Pacific: mirror of the Atlantic zones on the west coast
            limit = { has_global_flag = america_central_contact_pacific }
            every_province = { limit = { America_SmallpoxZone_WestCoast = yes } set_province_flag = america_smallpox_zone }
            America_SmallpoxSeedZone = { Type = 1 }
            every_province = { limit = { America_SmallpoxZone_PacificNorthMexico = yes } set_province_flag = america_smallpox_zone }
            America_SmallpoxSeedZone = { Type = 1 }
            every_province = { limit = { America_SmallpoxZone_PacificSouthMexico = yes } set_province_flag = america_smallpox_zone }
            America_SmallpoxSeedZone = { Type = 1 }
            every_province = { limit = { America_SmallpoxZone_PacificIsthmus = yes } set_province_flag = america_smallpox_zone }
            America_SmallpoxSeedZone = { Type = 1 }
            west_america_superregion = {
                set_key = { lhs = Plague_SpawnChance1 value = 2 }
            }
            log = "fire in the hole central america (pacific side)"
        }
        else = {
            every_province = { limit = { America_SmallpoxZone_Caribbean = yes } set_province_flag = america_smallpox_zone }
            America_SmallpoxSeedZone = { Type = 1 }
            every_province = { limit = { America_SmallpoxZone_GulfCoast = yes } set_province_flag = america_smallpox_zone }
            America_SmallpoxSeedZone = { Type = 1 }
            every_province = { limit = { America_SmallpoxZone_Yucatan = yes } set_province_flag = america_smallpox_zone }
            America_SmallpoxSeedZone = { Type = 1 }
            every_province = { limit = { America_SmallpoxZone_Isthmus = yes } set_province_flag = america_smallpox_zone }
            America_SmallpoxSeedZone = { Type = 1 }
            log = "fire in the hole central america"
        }
        clr_global_flag = america_smallpox_seeded
        central_america_superregion = {
            set_key = { lhs = Plague_SpawnChance1 value = 2 }
        }
        set_global_flag = active_america_plagues
        clr_global_flag = not_camerica_smallpox
        # South America follows a few years later
        country_event = { id = Plague_Spawner.2 days = 1825 random = 1825 }
    }
	
	option = {
		name = no_localisation
	}
}

	id = Plague_Spawner.3
	title = no_localisation
	desc = no_localisation
    picture = no_picture
	fire_only_once = yes
	is_triggered_only = yes
    hidden = yes
    
	trigger = {
        is_year = 1500
        FROM = {
        
            OR = {
                technology_group = andean
                technology_group = mesoamerican
                technology_group = south_american
                technology_group = andean
                technology_group = high_american
               
            }
            
            capital_scope = {
                superregion = central_america_superregion
            }
        }
        
        ROOT = {
            America_SalmonellaCarrier = yes
            OR = {
                AND = {
                    any_owned_province = {
                        is_core = FROM
                        superregion = central_america_superregion
                    }
                }

                AND = {
                    any_subject_country = {
                        any_owned_province = {
                            is_core = FROM
                            superregion = central_america_superregion
                        }
                    }
                }
            }
        }
    }

    immediate = {
        set_country_flag = salmonella_carrier
        # contact points: Plague_Spawner.4 seeds near these if no carrier-held seed exists
        every_owned_province = {
            limit = {
                is_core = FROM
                superregion = central_america_superregion
            }
            set_province_flag = salmonella_contact
        }
        every_subject_country = {
            every_owned_province = {
                limit = {
                    is_core = FROM
                    superregion = central_america_superregion
                }
                set_province_flag = salmonella_contact
            }
        }
    }

	option = {
		name = no_localisation
	}
    after = {
        if = { # one shared timer for conquest (.3) and settlement (.14); .4 fires on AAA so it can't be lost with the carrier
            limit = {
                NOT = { has_global_flag = salmonella_timer_started }
            }
            set_global_flag = salmonella_timer_started
            AAA = { country_event = { id = Plague_Spawner.4 days = 2555 random = 365 } }
        }
    }
}

				superregion = central_america_superregion
				NOT = { region = carribeans_region }
			}
		}
	}

	immediate = {
		every_country = {
			limit = {
				America_SalmonellaCarrier = yes

					superregion = central_america_superregion

				}
			}
			set_country_flag = salmonella_carrier
			every_owned_province = {
				limit = {
					superregion = central_america_superregion
					NOT = { region = carribeans_region }
				}

			}
		}

	}

		name = no_localisation

# Smallpox death toll timing: checked once a year, publishes (.17) once both the Central and the South American
# waves have hit and smallpox has died out in Central and South America. The South wave comes 5-10 years after
# the Central one, so the toll usually lands some 10-20 years after the first news.
# Safety: publishes anyway 25 years after counting started (e.g. a wave never fired, or recurrences keep smallpox alive).
country_event = {
	id = Plague_Spawner.23
	title = no_localisation
	desc = no_localisation
	picture = PLAGUE_eventPicture
	is_triggered_only = yes
	hidden = yes

	trigger = {
		tag = AAA
		has_global_flag = america_count_dead1 # stop if the toll was already published
	}

	immediate = {
		if = {
			limit = {
				OR = {
					AND = {
						NOT = { has_global_flag = not_camerica_smallpox } # Central wave has hit
						had_global_flag = { flag = america_smallpox_south_hit days = 730 } # South wave hit at least 2 years ago
						NOT = {
							any_province = {
								has_province_flag = Plague1
								OR = {
									superregion = central_america_superregion
									superregion = western_south_america_superregion
									superregion = eastern_south_america_superregion
								}
							}
						}
					}
					had_global_flag = { flag = america_count_dead1 days = 9125 } # 25 years
				}
			}
			country_event = { id = Plague_Spawner.17 }
		}
		else = {
			country_event = { id = Plague_Spawner.23 days = 365 }
		}
	}

	option = {
		name = no_localisation
	}
}
