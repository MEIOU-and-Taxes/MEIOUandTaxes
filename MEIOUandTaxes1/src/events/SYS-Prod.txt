#Find province that can upgrade
country_event = {
	id = POP_Prod.10
	title = POP_Prod.10.t
	desc = POP_Prod.10.d
	picture = UNIVERSITY_eventPicture

	is_triggered_only = yes
    hidden = yes
	
	immediate = {
        hidden_effect = {
			if = {
				limit = {
					innovativeness = 3.001
				}
				random_owned_province = { # kick off the process in one province, avoid multiple upgrades at the same time to avoid key corruption
					limit = {
						has_province_flag = Industry_CanUpgrade
					}
					province_event = { id = POP_Prod.11 days = 5 }
				}
			}
			else = {
				every_owned_province = {
					limit = {
						has_province_flag = Industry_CanUpgrade
					}
					add_permanent_province_modifier = {
						name = Urban_Innovativeness
						duration = 3650
					}
					clr_province_flag = Industry_CanUpgrade				
				}
			}
		}
	}
    option = {
        name = no_localisation
    }
}
