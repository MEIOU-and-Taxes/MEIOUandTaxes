andean_reciprocity_missions = {
	slot = 1
	generic = no
	ai = yes
	potential = {
		OR = {
			AND = {
				OR = {
					culture_group = andean_group
					culture_group = north_central_tradition_group
				}
				NOT = { tag = CZC }
				NOT = { tag = INC }
				NOT = { primary_culture = inca }
				NOT = { primary_culture = quechuan }
			}
			has_country_flag = andean_formed_tawantinsuyu
		}
	}

	andean_ties_of_reciprocity = {
		icon = mission_alliances
		required_missions = { }
		position = 1
		trigger = {
			custom_trigger_tooltip = {
				tooltip = andean_andean_ally_tt
				any_ally = {
					capital_scope = {
						OR = {
							region = peru_region
							region = bolivia_region
						}
					}
				}
			}
		}
		effect = {
			add_country_modifier = { name = andean_ties_of_reciprocity duration = 5475 }
			add_dip_power = 25
		}
	}

	andean_the_curacas = {
		icon = mission_mesoamerican_tributes
		required_missions = { andean_ties_of_reciprocity andean_lords_of_the_valley }
		position = 2

			custom_trigger_tooltip = {
				tooltip = andean_andean_subject_tt
				any_subject_country = {

						OR = {
							region = peru_region

						}
					}
				}
			}
		}
		effect = {
			add_country_modifier = { name = inca_curaca_reciprocity duration = 5475 }
			add_prestige = 10
		}
	}

	andean_lord_of_many_curacas = {
